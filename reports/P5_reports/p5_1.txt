"""P5-1 (second run): shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p5_1_msg.txt, list
"## Expected values", one line per name, `- NAME = value`; the item's date D from /tmp/p5_1_date.out (a.1).
"""
import hashlib
import os
import re
import subprocess

MSG = "/tmp/p5_1_msg.txt"
DATE_FILE = "/tmp/p5_1_date.out"


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    print(f"D (from {DATE_FILE}): {d}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    """Items separated by a space."""
    return value.split(" ")


def pairs(value):
    """`path:sha256` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in value.split(" | ")]


def arrows(value):
    """`kit name>repo path` items separated by ` | `."""
    return [tuple(item.split(">")) for item in value.split(" | ")]


REPO = path("~/Desktop/FX_Project/ofilt_pass1")


def git(*args):
    return subprocess.run(["git", "-C", REPO, *args], check=True,
                          capture_output=True).stdout.decode("utf-8")


def lines_of(data):
    return data.count(b"\n")


def access_log_hits():
    """Names starting ACCESS_LOG in REPO outside .git (names only; nothing opened, other names not printed)."""
    hits = []
    for root, dirs, files in os.walk(REPO):
        if root == REPO and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(root, n), REPO) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return hits


def porcelain():
    """`git status --porcelain --untracked-files=all` as a list of (status, path)."""
    out = git("status", "--porcelain", "--untracked-files=all")
    return [(ln[:2], ln[3:]) for ln in out.splitlines() if ln]


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")

#!/bin/bash
# P5-1 (second run) b.1: install the kit files and the two kept reports with cp -p. Runs no Python.
# Every path is read from the message's "## Expected values" list (KIT_DIR, INSTALL, P4_7C_SRC, P4_7C_DEST,
# P5_BOOT_SRC, P5_BOOT_DEST); `~` is expanded to the home folder.
set -u
MSG=/tmp/p5_1_msg.txt
REPO="$HOME/Desktop/FX_Project/ofilt_pass1"

val() {  # the value of `- NAME = value` in the Expected values list
  sed -n '/^## Expected values$/,$p' "$MSG" | sed -n "s/^- $1 = //p"
}
home() {  # expand a leading ~
  local v="$1"
  echo "${v/#\~/$HOME}"
}

KIT_DIR="$(home "$(val KIT_DIR)")"
rc=0
n=0
while IFS= read -r item; do
  [ -z "$item" ] && continue
  src="${item%%>*}"
  dest="${item#*>}"
  if cp -p "$KIT_DIR/$src" "$REPO/$dest"; then
    echo "cp -p \"$KIT_DIR/$src\" \"$REPO/$dest\": done"
    n=$((n + 1))
  else
    echo "cp -p \"$KIT_DIR/$src\" \"$REPO/$dest\": FAILED"
    rc=1
  fi
done < <(printf '%s\n' "$(val INSTALL)" | awk 'BEGIN { FS = " \\| " } { for (i = 1; i <= NF; i++) print $i }')
echo "kit files installed: $n"

if [ -d "$REPO/reports/P5_reports" ]; then
  echo "reports/P5_reports exists already"
else
  mkdir "$REPO/reports/P5_reports" && echo "mkdir reports/P5_reports: done" || rc=1
fi
for pair in "P4_7C_SRC P4_7C_DEST" "P5_BOOT_SRC P5_BOOT_DEST"; do
  set -- $pair
  src="$(home "$(val "$1")")"
  dest="$REPO/$(val "$2")"
  if cp -p "$src" "$dest"; then
    echo "cp -p \"$src\" \"$dest\": done"
  else
    echo "cp -p \"$src\" \"$dest\": FAILED"
    rc=1
  fi
done
echo "b.1 script exit status: $rc"
exit $rc

"""P5-1: shared helpers for the item's Python scripts.

"## Expected values", one line per name, `- NAME = value`; the item's date D from /tmp/p5_1_date.out (a.1).

import hashlib
import os
import re
import subprocess

MSG = "/tmp/p5_1_msg.txt"
DATE_FILE = "/tmp/p5_1_date.out"


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    """Items separated by a space."""
    return value.split(" ")


def pairs(value):
    """`path:sha256` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in value.split(" | ")]

def access_log_hits():
    """Names starting ACCESS_LOG in REPO outside .git (names only; nothing opened, other names not printed)."""
    hits = []
    for root, dirs, files in os.walk(REPO):
        if root == REPO and ".git" in dirs:

class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")
