"""P5-2: shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p5_2_msg.txt, list
"## Expected values", one line per name, `- NAME = value`; the item's date D from /tmp/p5_2_date.out (a.1).
"""
import hashlib
import os
import re
import subprocess

MSG = "/tmp/p5_2_msg.txt"
DATE_FILE = "/tmp/p5_2_date.out"


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    print(f"D (from {DATE_FILE}): {d}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    """Items separated by a space."""
    return value.split(" ")


def pairs(value):
    """`path:sha256` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in value.split(" | ")]


def arrows(value):
    """`kit name>repo path` items separated by ` | `."""
    return [tuple(item.split(">")) for item in value.split(" | ")]


REPO = path("~/Desktop/FX_Project/ofilt_pass1")


def git(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], check=True,
                          capture_output=True).stdout.decode("utf-8")


def lines_of(data):
    return data.count(b"\n")


def access_log_hits(root=REPO):
    """Names starting ACCESS_LOG under root outside .git (names only; nothing opened, other names not printed)."""
    hits = []
    for r, dirs, files in os.walk(root):
        if r == root and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(r, n), root) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return hits


def porcelain(repo=REPO):
    """`git status --porcelain --untracked-files=all` as a list of (status, path)."""
    out = git("status", "--porcelain", "--untracked-files=all", repo=repo)
    return [(ln[:2], ln[3:]) for ln in out.splitlines() if ln]


def logs_unchanged(C, E):
    for rel, key in (("BLIND_FIT_LOG", "BLIND_FIT_LOG"), ("RUN_REGISTRY.jsonl", "RUN_REGISTRY")):
        data = read_bytes(os.path.join(REPO, rel))
        C.check(sha256_bytes(data) == E[f"{key}_SHA256"] and lines_of(data) == int(E[f"{key}_LINES"]),
                f"{rel} sha256 == {key}_SHA256 and lines == {key}_LINES",
                f"{sha256_bytes(data)}, {lines_of(data)} lines")


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")

#!/bin/bash
# P5-2 b.1: install the kit files and P5-1's kept report with cp -p. Runs no Python.
# Every path is read from the message's "## Expected values" list (KIT_DIR, INSTALL, P5_1_SRC, P5_1_DEST);
# `~` is expanded to the home folder.
set -u
MSG=/tmp/p5_2_msg.txt
REPO="$HOME/Desktop/FX_Project/ofilt_pass1"

val() {  # the value of `- NAME = value` in the Expected values list
  sed -n '/^## Expected values$/,$p' "$MSG" | sed -n "s/^- $1 = //p"
}
home() {  # expand a leading ~
  local v="$1"
  echo "${v/#\~/$HOME}"
}

KIT_DIR="$(home "$(val KIT_DIR)")"
rc=0
n=0
while IFS= read -r item; do
  [ -z "$item" ] && continue
  src="${item%%>*}"
  dest="${item#*>}"
  if cp -p "$KIT_DIR/$src" "$REPO/$dest"; then
    echo "cp -p \"$KIT_DIR/$src\" \"$REPO/$dest\": done"
    n=$((n + 1))
  else
    echo "cp -p \"$KIT_DIR/$src\" \"$REPO/$dest\": FAILED"
    rc=1
  fi
done < <(printf '%s\n' "$(val INSTALL)" | awk 'BEGIN { FS = " \\| " } { for (i = 1; i <= NF; i++) print $i }')
echo "kit files installed: $n"

src="$(home "$(val P5_1_SRC)")"
dest="$REPO/$(val P5_1_DEST)"
if cp -p "$src" "$dest"; then
  echo "cp -p \"$src\" \"$dest\": done"
else
  echo "cp -p \"$src\" \"$dest\": FAILED"
  rc=1
fi
echo "b.1 script exit status: $rc"
exit $rc

"""P5-2 b.2: PASS5_STATUS.md rewritten from the kit's template, with D (verdict line `B2 CHECK`).

Imports no module of ofilt; reads and writes files as bytes. Writes one file, REPO/PASS5_STATUS.md, and only if it
has BASE_STATUS_SHA256 (as P5-1 committed it) and the template passes its checks.
"""

#!/bin/bash
# P5-2 f: the tagged-clone rehearsal (P5Q4). Runs no Python itself; the unittest command runs in the clone.
# CLONE, EMPTY and REHEARSAL are read from the message's "## Expected values" list (not typed); the commit to rehearse
# is step e's, read from /tmp/p5_2_e.out. It prints `key=value` facts that /tmp/p5_2_f_check.py checks. A temporary tag
# exists only inside the clone, whose remote is removed before the tag is made; the clone and EMPTY are removed at the
# end, on success or failure, once this script has created them.
set -u
MSG=/tmp/p5_2_msg.txt
REPO="$HOME/Desktop/FX_Project/ofilt_pass1"
val() {  # the value of `- NAME = value` in the Expected values list
  sed -n '/^## Expected values$/,$p' "$MSG" | sed -n "s/^- $1 = //p"
}
CLONE="$(val CLONE)"
EMPTY="$(val EMPTY)"
REHEARSAL="$(val REHEARSAL)"
E_HEAD="$(sed -n 's/^commit \([0-9a-f]\{40\}\)$/\1/p' /tmp/p5_2_e.out)"
echo "CLONE=$CLONE"
echo "EMPTY=$EMPTY"
echo "REHEARSAL=$REHEARSAL"
echo "e_head=$E_HEAD"
for p in "$CLONE" "$EMPTY"; do      # only these two paths, both under /tmp/p5_2_
  case "$p" in
    /tmp/p5_2_?*) ;;
    *) echo "REFUSED: $p is not under /tmp/p5_2_"; exit 2 ;;
  esac
done
if ! printf '%s' "$REHEARSAL" | grep -Eq '^tests\.[A-Za-z0-9_.]+( tests\.[A-Za-z0-9_.]+)*$'; then
  echo "REFUSED: REHEARSAL is not a list of tests: '$REHEARSAL'"; exit 2
fi
if ! printf '%s' "$E_HEAD" | grep -Eq '^[0-9a-f]{40}$'; then
  echo "REFUSED: no commit line in /tmp/p5_2_e.out"; exit 2
fi

# 1. CLONE and EMPTY must not exist; EMPTY is created empty.
if [ -e "$CLONE" ] || [ -L "$CLONE" ] || [ -e "$EMPTY" ] || [ -L "$EMPTY" ]; then
  echo "step1_absent_before=no"
  echo "REFUSED: $CLONE or $EMPTY exists; nothing created, nothing removed"
  exit 1
fi
echo "step1_absent_before=yes"
cleanup() {  # 8. remove the two paths, and only them
  rm -rf "$CLONE" "$EMPTY"
  if [ -e "$CLONE" ] || [ -L "$CLONE" ]; then echo "step8_clone_gone=no"; else echo "step8_clone_gone=yes"; fi
  if [ -e "$EMPTY" ] || [ -L "$EMPTY" ]; then echo "step8_empty_gone=no"; else echo "step8_empty_gone=yes"; fi
}
trap cleanup EXIT
mkdir "$EMPTY"
echo "step1_empty_entries=$(ls -A "$EMPTY" | wc -l | tr -d ' ')"

# 2. The clone, without its remote; its HEAD must be step e's commit.
echo "step2_repo_head=$(git -C "$REPO" rev-parse HEAD)"
git clone -q --no-hardlinks "$REPO" "$CLONE" || { echo "step2_clone=failed"; exit 1; }
git -C "$CLONE" remote remove origin || { echo "step2_remote_remove=failed"; exit 1; }
echo "step2_clone_remotes=$(git -C "$CLONE" remote | wc -l | tr -d ' ')"
CLONE_HEAD="$(git -C "$CLONE" rev-parse HEAD)"
echo "step2_clone_head=$CLONE_HEAD"
echo "step2_clone_tags_before=$(git -C "$CLONE" tag -l | tr '\n' ' ' | sed 's/ $//')"
if [ "$CLONE_HEAD" != "$E_HEAD" ]; then
  echo "REFUSED: the clone's HEAD is not step e's commit; no tag made"; exit 1
fi

# 3. The local annotated tag, in the clone only.
git -C "$CLONE" -c user.name=rehearsal -c user.email=rehearsal@localhost tag -a prereg-v1 \
  -m "P5-2 rehearsal (P5Q4); local to this clone" || { echo "step3_tag=failed"; exit 1; }
echo "step3_clone_tags_after=$(git -C "$CLONE" tag -l | tr '\n' ' ' | sed 's/ $//')"
echo "step3_tag_type=$(git -C "$CLONE" cat-file -t refs/tags/prereg-v1)"
echo "step3_tag_commit=$(git -C "$CLONE" rev-parse 'refs/tags/prereg-v1^{commit}')"

# 4. The rehearsal in the clone, with OFILT_DATA_DIR at EMPTY.
CMD='source "$HOME/venvs/ofilt_p2/bin/activate" && cd "'"$CLONE"'" && export OFILT_DATA_DIR="'"$EMPTY"'" && python -B -m unittest '"$REHEARSAL"' 2>&1 | tee /tmp/p5_2_f.out'
echo "step4_command=env -u OFILT_RUN_SEALED bash -c '$CMD'"
echo "---- step 4 output ----"
env -u OFILT_RUN_SEALED bash -c "$CMD"
echo "---- end of step 4 output ----"

echo "step6_clone_access_log_names=$(find "$CLONE" -path "$CLONE/.git" -prune -o -name 'ACCESS_LOG*' -print | wc -l | tr -d ' ')"
# 7. REPO untouched.
echo "step7_repo_head=$(git -C "$REPO" rev-parse HEAD)"
echo "step7_repo_status_lines=$(git -C "$REPO" status --porcelain --untracked-files=all | wc -l | tr -d ' ')"
