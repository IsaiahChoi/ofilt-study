def show(rel, a, b):

def full(rel):
    b = (REPO / rel).read_bytes()

DEFS = {}


def defs(rel):
    """[(start, end, qualname, kind)] for every function and class of a Python file (start includes decorators)."""
    if rel in DEFS:
        return DEFS[rel]
    out = []

    def visit(node, prefix):
        for ch in ast.iter_child_nodes(node):
            if isinstance(ch, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                start = min([d.lineno for d in ch.decorator_list] + [ch.lineno])
                q = f"{prefix}.{ch.name}" if prefix else ch.name
                out.append((start, ch.end_lineno, q, "class" if isinstance(ch, ast.ClassDef) else "def"))
                visit(ch, q)
            else:
                visit(ch, prefix)

    visit(ast.parse((REPO / rel).read_text(encoding="utf-8")), "")
    DEFS[rel] = out
    return out


def enclosing(rel, i):
    if not rel.endswith(".py"):

    return min(inside, key=lambda d: d[1] - d[0])[2]

def defs(rel):

def tree_of(rel):
    if rel not in TREES:

def bindings(rel):
    out = {}
    pkg = ".".join(rel.split("/")[:-1])
    for node in ast.walk(tree_of(rel)):
        if isinstance(node, ast.Import):
            for a in node.names:
                if a.name in INDEX:
                    out[a.asname or a.name] = ("mod", a.name)
        elif isinstance(node, ast.ImportFrom):
            base = node.module or ""
            if node.level:
                up = pkg.split(".")[: len(pkg.split(".")) - (node.level - 1)]
                base = ".".join(up + ([node.module] if node.module else []))
            for a in node.names:
                full = f"{base}.{a.name}" if base else a.name
                if full in INDEX:
                    out[a.asname or a.name] = ("mod", full)
                else:
                    out[a.asname or a.name] = ("fn", base, a.name)
    return out


def calls(rel):
    """[(line, enclosing def name or None, (module, function))] for every resolvable call in a file."""
    me, B, own = modname(rel), bindings(rel), {d[2].split(".")[-1] for d in defs(rel) if d[3] == "def"}
    out = []
    for node in ast.walk(tree_of(rel)):
        if not isinstance(node, ast.Call):
            continue
        f, tgt = node.func, None
        if isinstance(f, ast.Name):
            b = B.get(f.id)
            if b and b[0] == "fn":
                tgt = (b[1], b[2])
            elif f.id in own:
                tgt = (me, f.id)
        elif isinstance(f, ast.Attribute) and isinstance(f.value, ast.Name):
            b = B.get(f.value.id)
            if b and b[0] == "mod":
                tgt = (b[1], f.attr)
            elif f.value.id in ("self", "cls"):
                tgt = (me, f.attr)
        if tgt:
            enc = [d for d in defs(rel) if d[3] == "def" and d[0] <= node.lineno <= d[1]]
            name = min(enc, key=lambda d: d[1] - d[0])[2].split(".")[-1] if enc else None
            out.append((node.lineno, name, tgt))
    return out
