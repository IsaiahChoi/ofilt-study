# ------------------------------------------------------------------ the supervisor's texts
def blocks():
    """{name: [lines]} for HEADER, AM-07, AM-08, AM-09, AM-10 and PLAN; title lines kept under 'title'."""
    out, cur = {}, None
    for line in TEXTS.splitlines():
        m = re.match(r"^(HEADER|PLAN)$|^(AM-\d\d) \((.*)\)$", line)
        if m:
            cur = m.group(1) or m.group(2)
            out[cur] = {"title": line, "lines": []}
            continue
        out[cur]["lines"].append(line)
    for b in out.values():                                   # drop the blank lines that separate the blocks
        while b["lines"] and not b["lines"][-1].strip():
            b["lines"].pop()
    return out


def amendment(am):
    """heading, texts {variant or None: text}, lines [(n, variant, template)], placeholders (or None)."""
    b = blocks()[am]
    a = {"id": am, "title": b["title"], "heading": None, "texts": {}, "lines": [], "placeholders": None}
    for e in b["lines"]:
        if e.startswith("Heading: "):
            a["heading"] = e[len("Heading: "):]
        elif e.startswith("Placeholders: "):
            a["placeholders"] = e
        elif e.startswith("Text"):
            m = re.match(r'^Text(?:, variant "([^"]+)")?: (.*)$', e)
            a["texts"][m.group(1)] = m.group(2)
        elif e.startswith("Line "):
            m = re.match(r'^Line (\d+)(?:, variant "([^"]+)")?: ', e)
            a["lines"].append((int(m.group(1)), m.group(2), e.split(": ", 1)[1]))  # the text after the first ": "
        else:
            raise SystemExit(f"{am}: unparsed entry {e[:60]!r}")

    def once(s, a, b):

def blocks():
    """{name: [lines]} for HEADER, AM-07, AM-08, AM-09, AM-10 and PLAN; title lines kept under 'title'."""
    out, cur = {}, None
    for line in TEXTS.splitlines():
        m = re.match(r"^(HEADER|PLAN)$|^(AM-\d\d) \((.*)\)$", line)
        if m:
            cur = m.group(1) or m.group(2)
            out[cur] = {"title": line, "lines": []}
            continue
        out[cur]["lines"].append(line)
    for b in out.values():                                   # drop the blank lines that separate the blocks
        while b["lines"] and not b["lines"][-1].strip():
            b["lines"].pop()
    return out


def amendment(am):
    """heading, texts {variant or None: text}, lines [(n, variant, template)], placeholders (or None)."""
    b = blocks()[am]
    a = {"id": am, "title": b["title"], "heading": None, "texts": {}, "lines": [], "placeholders": None}
    for e in b["lines"]:
        if e.startswith("Heading: "):
            a["heading"] = e[len("Heading: "):]
        elif e.startswith("Placeholders: "):
            a["placeholders"] = e
        elif e.startswith("Text"):
            m = re.match(r'^Text(?:, variant "([^"]+)")?: (.*)$', e)
            a["texts"][m.group(1)] = m.group(2)
        elif e.startswith("Line "):
            m = re.match(r'^Line (\d+)(?:, variant "([^"]+)")?: ', e)
            a["lines"].append((int(m.group(1)), m.group(2), e.split(": ", 1)[1]))  # the text after the first ": "
        else:
            raise SystemExit(f"{am}: unparsed entry {e[:60]!r}")
    return a

def edit_specpy(src, am, variant=None):
    def once(s, a, b):
        if s.count(a) != 1:
            raise SystemExit(f"spec.py anchor found {s.count(a)} times: {a[:60]!r}")
        return s.replace(a, b)
    if am == "AM-08":
        src = once(src, ANCHOR_INF, ANCHOR_INF + REF_CONST[variant])
        return once(src, PINNED_REF, '              "reference": REFERENCE, "test": "two_sided"}  # AM-08\n')

def fill(text, method):
    v = standins()
    vals = {k: v[k] for k in ("S", "SL", "SH", "C95", "C90", "CCW", "R_dk", "R_boot", "R_cw")}
    vals.update(v["by_m"][method], M=method, date=DATE)
    out = PH.sub(lambda m: vals[m.group(0)[1:-1]] if m.group(0)[1:-1] in vals else m.group(0), text)
    left = PH.findall(out)
    if left:
        raise SystemExit(f"unfilled placeholders {left}")
    return out

def apply(clone, branch):
    clone = Path(clone).resolve()
    if not str(clone).startswith(("/tmp/p3_3e_clone", "/private/tmp/p3_3e_clone")):
        raise SystemExit(f"REFUSED: {clone} is not a /tmp/p3_3e_clone* scratch clone")
    br = BRANCHES[branch]
    method = br["method"]
    print(f"dry switch {branch}: {br['desc']}; <M> = {method}; <date> = {DATE}")
    sl = spec_lines(clone)
    n_before = len(sl)
    amd = (clone / AMD_REL).read_text(encoding="utf-8")
    for am, variant in br["ams"]:
        n, old, new, sec = section_text(am, variant, method)

        sl[n - 1] = new

        print(f"  {am}{' (' + variant + ')' if variant else ''}: line {n} replaced; section appended")

            d = fill(d, method)
            p = Path(f"/tmp/p3_3e_{branch}_{am}.diff")
            p.write_text(d, encoding="utf-8")
            for args in (["apply", "--check", str(p)], ["apply", str(p)]):
                r = subprocess.run(["git", "-C", str(clone), *args], capture_output=True, text=True)

                    raise SystemExit("git apply failed")

    (clone / AMD_REL).write_text(amd, encoding="utf-8")
    print(f"  spec line count: {n_before} before, {len(spec_lines(clone))} after; equal: {n_before == len(spec_lines(clone))}")
