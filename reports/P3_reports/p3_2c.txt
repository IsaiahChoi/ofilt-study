   213	    """Problems with a configuration: missing keys, or any difference from the configuration rebuilt now."""
   214	    missing = [k for k in REQUIRED_KEYS if k not in cfg]
   215	    if missing:
   216	        return [f"missing key(s) {missing}"]
   217	    rebuilt = build_config()
   218	    return [f"key {k!r} differs from the registry" for k in sorted(set(rebuilt) | set(cfg)) if cfg.get(k) != rebuilt.get(k)]

   221	def combo_refusal(*, real: bool, smoke, rehearsal: bool) -> str | None:
   222	    """Mode combinations, checked first (before any import of ofilt or any data)."""
   223	    if real and smoke is not None:
   224	        return "--real cannot be combined with --smoke"
   225	    if real and rehearsal:
   226	        return "--real cannot be combined with --rehearsal"
   227	    if smoke is not None and smoke < 2:
   228	        return "--smoke needs N >= 2"
   229	    return None

   231	
   232	def mode_refusal(*, real: bool, smoke, rehearsal: bool, sealed_value, config_sha: str, expected_sha, dirty) -> str | None:
   233	    """The reason to refuse, or None: the mode combinations, then, in real mode, the sealed environment variable,
   234	    the expected configuration SHA-256 and the tree, in that order. It reads nothing itself."""
   235	    reason = combo_refusal(real=real, smoke=smoke, rehearsal=rehearsal)
   236	    if reason or not real:
   237	        return reason
   238	    if sealed_value != "1":
   239	        return f"--real needs {SEALED_ENV}=1 for this command"
   240	    if expected_sha is None or expected_sha != config_sha:
   241	        return "--expect-config-sha256 does not equal the configuration file's SHA-256"
   242	    if dirty:
   243	        return "the working tree is dirty"
   244	    return None

   247	def check_log(path: Path, n_expected: int, panel_sha: str | None = None, caller: str | None = None) -> list:
   248	    """Problems with BLIND_FIT_LOG: line count, all dev_real; and, if given, the last line's panel hash and caller."""
   249	    recs = [json.loads(x) for x in Path(path).read_text(encoding="utf-8").splitlines() if x.strip()]
   250	    p = []
   251	    if len(recs) != n_expected:
   252	        p.append(f"{len(recs)} lines, expected {n_expected}")
   253	    if any(r.get("provenance") != "dev_real" for r in recs):
   254	        p.append("a line is not dev_real")
   255	    if recs and panel_sha is not None and recs[-1].get("panel_sha256") != panel_sha:
   256	        p.append("the last line's panel hash differs")
   257	    if recs and caller is not None and caller not in str(recs[-1].get("caller", "")):
   258	        p.append("the last line's caller does not name this script")
   259	    return p

   262	def seed_ranges(cfg: dict) -> list:
   263	    """(lo, hi) offsets from the spec seed, inclusive, of every configured seed (outer and warp)."""
   264	    s = cfg["seeds"]
   265	    out = []
   266	    for c in cfg["cells"]:
   267	        lo = s["outer_offset"] + s["cell_stride"] * c["id"]
   268	        out.append((lo, lo + c["replications"] - 1))
   269	        if c["id"] in cfg["pooled_cells"]:
   270	            out.append((lo + s["warp_offset"], lo + s["warp_offset"] + c["replications"] - 1))
   271	    return out

   274	def check_output(obj, allowed: set, path: str = "") -> list:
   275	    """Problems: keys outside the whitelist, or lists longer than MAX_LIST."""
   276	    p = []
   277	    if isinstance(obj, dict):
   278	        for k, v in obj.items():
   279	            if k not in allowed:
   280	                p.append(f"key {path}/{k} not in the whitelist")
   281	            p += check_output(v, allowed, f"{path}/{k}")
   282	    elif isinstance(obj, list):
   283	        if len(obj) > MAX_LIST:
   284	            p.append(f"list {path} has {len(obj)} entries")
   285	        for i, v in enumerate(obj):
   286	            p += check_output(v, allowed, f"{path}[{i}]")
   287	    return p

   290	def _script_line(e: BaseException) -> int:
   291	    tb = traceback.extract_tb(e.__traceback__)
   292	    mine = [f for f in tb if f.filename.endswith(SCRIPT_NAME)]
   293	    return int((mine or tb)[-1].lineno) if tb else -1

   296	# ====================================================================== the run
   297	def main(argv: list | None = None) -> int:
   298	    ap = argparse.ArgumentParser()
   299	    g = ap.add_mutually_exclusive_group()
   300	    g.add_argument("--synthetic", action="store_true")
   301	    g.add_argument("--real", action="store_true")
   302	    ap.add_argument("--rehearsal", action="store_true")
   303	    ap.add_argument("--smoke", type=int, default=None)
   304	    ap.add_argument("--expect-config-sha256", default=None)
   305	    ap.add_argument("--out-dir", type=Path, default=None)
   306	    ap.add_argument("--write-config", action="store_true")
   307	    ap.add_argument("--check-config", action="store_true")
   308	    a = ap.parse_args(argv)
   309	    if a.write_config or a.check_config:
   310	        cfg = build_config()
   311	        if a.write_config:
   312	            if CONFIG_PATH.exists():
   313	                print(f"REFUSED: {CONFIG_PATH} exists")
   314	                return EXIT_REFUSED
   315	            _write_file(CONFIG_PATH, config_text(cfg))
   316	        problems = validate_config(json.loads(CONFIG_PATH.read_text(encoding="utf-8")))
   317	        print(f"config {CONFIG_PATH.relative_to(ROOT)} SHA-256 {sha256_file(CONFIG_PATH)}; valid: {not problems}")
   318	        return 0 if not problems else EXIT_REFUSED
   319	    reason = combo_refusal(real=a.real, smoke=a.smoke, rehearsal=a.rehearsal)
   320	    if reason:
   321	        print(f"REFUSED: {reason}")
   322	        return EXIT_REFUSED
   323	    sys_tmp = Path(tempfile.gettempdir()).resolve()
   324	    if a.real:
   325	        if a.out_dir is not None:
   326	            print("REFUSED: --real writes reports/P3_4_sealed_run.* and takes no --out-dir")
   327	            return EXIT_REFUSED
   328	        out_dir = REAL_OUT
   329	    else:
   330	        if a.out_dir is None:
   331	            print("REFUSED: synthetic modes need --out-dir under /tmp")
   332	            return EXIT_REFUSED
   333	        out_dir = a.out_dir.resolve()
   334	        if not (out_dir.is_relative_to(Path("/tmp").resolve()) or out_dir.is_relative_to(sys_tmp)):
   335	            print("REFUSED: --out-dir must lie under /tmp (or the system temporary directory)")
   336	            return EXIT_REFUSED
   337	    # (1) the run's own temporary directory, before any ofilt import (DQ1.2)
   338	    run_tmp = Path(tempfile.mkdtemp(prefix="p3_sealed_run_")).resolve()
   339	    previous = tempfile.tempdir
   340	    tempfile.tempdir = str(run_tmp)
   341	    try:
   342	        return _run(a, run_tmp, out_dir)
   343	    except Refused as e:
   344	        print(f"REFUSED: {e}")
   345	        return EXIT_REFUSED
   346	    except AbortRun as e:
   347	        print(f"ABORTED at line {_script_line(e)} (before the first flush)")
   348	        return EXIT_ABORTED
   349	    finally:
   350	        shutil.rmtree(run_tmp, ignore_errors=True)
   351	        tempfile.tempdir = previous

   354	def _run(a, run_tmp: Path, out_dir: Path) -> int:
   355	    import importlib
   356	    R = importlib.import_module("ofilt.drift.registry")
   357	    if Path(R.SYNTHETIC_PATH).resolve().parent != run_tmp:
   358	        raise AbortRun("registry.SYNTHETIC_PATH lies outside the run's temporary directory")
   359	    import numpy as np
   360	    from scipy import stats
   361	    from ofilt.drift import calib as C
   362	    from ofilt.drift import fctests as FT
   363	    from ofilt.drift import forecast as F
   364	    from ofilt.drift import inference as I
   365	    from ofilt.drift import io as IO
   366	    from ofilt.drift import sim as S
   367	    from ofilt.drift import targets as TG
   368	    from ofilt.drift.models import M2, M3
   369	    from ofilt.drift.spec import Spec
   370	    from p3_0b_preflight import verify_panel
   371	
   372	    # (2) configuration and mode conditions, before any data file is read
   373	    cfg_bytes = CONFIG_PATH.read_bytes()
   374	    cfg_sha = sha256_bytes(cfg_bytes)
   375	    cfg = json.loads(cfg_bytes.decode("utf-8"))
   376	    problems = validate_config(cfg)
   377	    if problems:
   378	        raise Refused(f"configuration: {problems}")
   379	    head, dirty = R.git_state()
   380	    if a.real:
   381	        reason = mode_refusal(real=True, smoke=a.smoke, rehearsal=a.rehearsal,
   382	                              sealed_value=os.environ.get(SEALED_ENV), config_sha=cfg_sha,
   383	                              expected_sha=a.expect_config_sha256, dirty=dirty)
   384	        if reason:
   385	            raise Refused(reason)
   386	    spec = Spec.load()
   387	    alpha = float(cfg["methods"]["alpha"])
   388	    z2, z1 = float(stats.norm.ppf(1 - alpha / 2)), float(stats.norm.ppf(1 - alpha))
   389	    lag6, h1n = int(cfg["expected_lag"]), spec.h1_parameter
   390	    mode = "real" if a.real else ("rehearsal" if a.rehearsal else "synthetic")
   391	    paths = (out_dir / f"{REAL_STEM}.json", out_dir / f"{REAL_STEM}.md")
   392	    allowed = set(cfg["outputs"])
   393	    t_start = time.time()
   394	    state = {"tool": f"tools/{SCRIPT_NAME}", "mode": mode, "smoke": a.smoke, "config_sha256": cfg_sha,
   395	             "spec_sha256": cfg["spec_sha256"], "dev_panel_sha256": cfg["dev_panel_sha256"], "head": head,
   396	             "calibrator": None, "cells": [], "pooled": None, "mde80": None, "t3": None, "errors": [],
   397	             "complete": False, "aborted": None, "runtime": {"tasks": [], "calibrator_seconds": None,
   398	                                                              "total_seconds": None}}
   399	
   400	    def flush():
   401	        state["runtime"]["total_seconds"] = time.time() - t_start
   402	        p = check_output(state, allowed)
   403	        if p:
   404	            raise AbortRun(f"output check: {p[:3]}")
   405	        _write_file(paths[0], json.dumps(state, indent=1, sort_keys=True, allow_nan=False) + "\n")
   406	        _write_file(paths[1], render_md(state))
   407	
   408	    def assert_lag(lag, what):
   409	        if int(lag) != lag6:
   410	            raise AbortRun(f"{what}: lag {lag} != {lag6}")
   411	
   412	    try:
   413	        # ------------------------------------------------ data and the ONE calibrator
   414	        if a.real:
   415	            if spec.sha256 != cfg["spec_sha256"]:
   416	                raise AbortRun("spec SHA-256 differs from the configuration")
   417	            digest = verify_panel(IO.DATA_DIR / IO.DEV_FILE, spec)
   418	            if digest != cfg["dev_panel_sha256"]:
   419	                raise AbortRun("dev panel SHA-256 differs from the configuration")
   420	            base = TG.add_targets(IO.load_dev(IO.DATA_DIR / IO.DEV_FILE, spec), spec)
   421	            assert_am03_target(base, spec)
   422	            log_path = S.BLIND_FIT_LOG_PATH
   423	            p = check_log(log_path, 6)
   424	            if p:
   425	                raise AbortRun(f"BLIND_FIT_LOG before: {p}")
   426	        else:
   427	            real_log = ROOT / "BLIND_FIT_LOG"
   428	            real_log_sha = sha256_file(real_log)
   429	            S.BLIND_FIT_LOG_PATH = run_tmp / "BLIND_FIT_LOG"
   430	            if a.rehearsal:
   431	                digest = verify_panel(IO.DATA_DIR / IO.DEV_FILE, spec)
   432	                if digest != cfg["dev_panel_sha256"]:
   433	                    raise AbortRun("dev panel SHA-256 differs from the configuration")
   434	                dev = TG.add_targets(IO.load_dev(IO.DATA_DIR / IO.DEV_FILE, spec), spec)   # registers real targets
   435	                assert_am03_target(dev, spec)
   436	                del dev
   437	            base = stand_in(cfg, spec, S, np)
   438	        t0 = time.time()
   439	        cal = S.DevCalibrator(base, spec)                                   # the ONE sealed fit
   440	        state["runtime"]["calibrator_seconds"] = time.time() - t0
   441	        del base
   442	        if a.real:
   443	            p = check_log(log_path, 7, panel_sha=cfg["dev_panel_sha256"], caller=SCRIPT_NAME)
   444	            if p:
   445	                raise AbortRun(f"BLIND_FIT_LOG after the calibrator: {p}")
   446	        assert_calibrator_shape(cal, cfg)
   447	        state["calibrator"] = {"T_source": int(cal.T), "rows": int(cfg["rows"])}
   448	        flush()
   449	
   450	        cells = {c["id"]: c for c in cfg["cells"]}
   451	        smoke = a.smoke
   452	
   453	        def reps(c):
   454	            return c["replications"] if smoke is None else min(c["replications"], smoke)
   455	
   456	        def seed(c, r):
   457	            s = cfg["seeds"]
   458	            return spec.seed + s["outer_offset"] + s["cell_stride"] * c["id"] + r
   459	
   460	        mem = {}                                                            # per-replication values, memory only
   461	
   462	        def run_task(name, fn, n_cells):
   463	            t = time.time()
   464	            n_reps = 0
   465	            try:
   466	                n_reps = fn()
   467	            except AbortRun:
   468	                raise
   469	            except Exception as e:                                          # class and line only (DQ1.4)
   470	                state["errors"].append({"task": name, "error_class": type(e).__name__, "line": _script_line(e)})
   471	            sec = time.time() - t
   472	            state["runtime"]["tasks"].append({"task": name, "seconds": sec, "replications": n_reps,
   473	                                              "seconds_per_replication": sec / n_reps if n_reps else None})
   474	            flush()
   475	
   476	        # ------------------------------------------------ cells 0 to 4 (T1, T5 boundary, T2)
   477	        def h1_cell(cid):
   478	            c = cells[cid]
   479	            n = reps(c)
   480	            v = {k: np.empty(n) for k in ("b", "se", "t", "t3", "ts", "ts3")}
   481	            for r in range(n):
   482	                s = seed(c, r)
   483	                y = cal.draw(c["b"], float(cfg["a2"]), seed=s, n_dates=c["n_dates"])
   484	                f = C.fit_t(y, spec, "y_sim")
   485	                assert_lag(f["lag"], f"cell {cid} fit")
   486	                w = C.bootstrap_t_given_fit(y, spec, np.random.default_rng(s + cfg["seeds"]["warp_offset"]),
   487	                                            "y_sim", 1, f, cfg["mean_block"])
   488	                v["b"][r], v["se"][r], v["t"][r], v["t3"][r] = f["b"], f["se"], f["t"], f["t_h3"]
   489	                v["ts"][r], v["ts3"][r] = w["t_star"][0], w["t_star_h3"][0]
   490	            mem[cid] = v
   491	            out = cell_head(c, n)
   492	            out["estimates"] = est_summary(v["b"], v["se"], c["b"])
   493	            out["dk_normal"] = method_block(v, c["b"], z2, z1, cfg, cid, C)
   494	            if cid == 0:
   495	                out["h3"] = {"dk_normal": rate(int((np.abs(v["t3"]) > z2).sum()), n, C), "bootstrap": None}
   496	            state["cells"].append(out)
   497	            return n
   498	
   499	        def task_cells(ids):
   500	            def fn():
   501	                tot = 0
   502	                for cid in ids:
   503	                    tot += h1_cell(cid)
   504	                    flush()
   505	                return tot
   506	            return fn
   507	
   508	        run_task("T1", task_cells([0]), 1)
   509	        run_task("T5_boundary", task_cells([1]), 1)
   510	        run_task("T2", task_cells([2, 3, 4]), 3)
   511	
   512	        # ------------------------------------------------ pooled critical values (D3.2, D3.3 amended)
   513	        def pooled():
   514	            ids = cfg["pooled_cells"]
   515	            if not all(i in mem for i in ids):
   516	                raise RuntimeError("pooled cells incomplete")
   517	            ts = np.abs(np.concatenate([mem[i]["ts"] for i in ids]))
   518	            zc = np.abs(np.concatenate([(mem[i]["b"] - cells[i]["b"]) / mem[i]["se"] for i in ids]))
   519	            ts3 = np.abs(np.concatenate([mem[i]["ts3"] for i in ids]))
   520	            pool = {"cells": list(ids), "c_boot": qpair(ts, C), "c_cal": qpair(zc, C),
   521	                    "c_boot_h3": {"q95": os_summary(ts3, 0.95, C)}}
   522	            state["pooled"] = pool
   523	            cb95, cb90 = pool["c_boot"]["q95"]["value"], pool["c_boot"]["q90"]["value"]
   524	            cc95, cc90 = pool["c_cal"]["q95"]["value"], pool["c_cal"]["q90"]["value"]
   525	            for out in state["cells"]:
   526	                cid = out["cell"]
   527	                if cid not in ids:
   528	                    continue
   529	                v, b = mem[cid], cells[cid]["b"]
   530	                out["bootstrap"] = method_block(v, b, cb95, cb90, cfg, cid, C)
   531	                out["calibrated"] = method_block(v, b, cc95, cc90, cfg, cid, C)
   532	                out["own_quantiles"] = {"c_boot": qpair(np.abs(v["ts"]), C),
   533	                                        "c_cal": qpair(np.abs((v["b"] - b) / v["se"]), C)}
   534	                if cid == 0:
   535	                    out["h3"]["bootstrap"] = rate(int((np.abs(v["t3"]) > pool["c_boot_h3"]["q95"]["value"]).sum()),
   536	                                                  len(v["t3"]), C)
   537	            mde_ids = [0, 2, 3, 4]
   538	            sd = float(np.std(np.concatenate([mem[i]["b"] - cells[i]["b"] for i in ids]), ddof=1))
   539	            mse = float(np.mean(np.concatenate([mem[i]["se"] for i in ids])))
   540	            by = {o["cell"]: o for o in state["cells"]}
   541	            state["mde80"] = {"cells": mde_ids}
   542	            for m in ("dk_normal", "bootstrap", "calibrated"):
   543	                state["mde80"][m] = C.mde80([cells[i]["b"] for i in mde_ids],
   544	                                            [by[i][m]["reject"]["rate"] for i in mde_ids], sd=sd, mean_se=mse)
   545	            return 0
   546	
   547	        run_task("pooled", pooled, 0)
   548	
   549	        # ------------------------------------------------ T6 (cells 5 to 8)
   550	        def t6():
   551	            tot = 0
   552	            for cid in (5, 6, 7, 8):
   553	                c = cells[cid]
   554	                n = reps(c)
   555	                v = {k: np.empty(n) for k in ("ob", "ose", "hb", "hse", "hsh", "wb", "wse")}
   556	                for r in range(n):
   557	                    s = seed(c, r)
   558	                    if c["window"] == "early":
   559	                        pos = C.window_positions(cfg["K_early"], c["n_dates"], np.random.default_rng(s),
   560	                                                 cfg["mean_block"])
   561	                        y = cal.draw(c["b"], float(cfg["a2"]), seed=s, positions=pos, n_dates=c["n_dates"])
   562	                    else:
   563	                        y = cal.draw(c["b"], float(cfg["a2"]), seed=s, n_dates=c["n_dates"])
   564	                    f = C.fit_t(y, spec, "y_sim")
   565	                    h = I.fit_huber(I.build_design(y, spec, target="y_sim"), spec)
   566	                    fw = C.fit_t(C.winsorise_fx(y, spec), spec, "y_sim")
   567	                    assert_lag(f["lag"], f"cell {cid} OLS")
   568	                    assert_lag(h.lag, f"cell {cid} Huber")
   569	                    assert_lag(fw["lag"], f"cell {cid} winsorised")
   570	                    ph = h.param(h1n)
   571	                    v["ob"][r], v["ose"][r] = f["b"], f["se"]
   572	                    v["hb"][r], v["hse"][r], v["hsh"][r] = ph["estimate"], ph["se"], h.info["share_downweighted"]
   573	                    v["wb"][r], v["wse"][r] = fw["b"], fw["se"]
   574	                out = cell_head(c, n)
   575	                out["estimators"] = {
   576	                    "ols_dk": t6_block(v["ob"], v["ose"], c["b"], z2, C),
   577	                    "huber_dk": dict(t6_block(v["hb"], v["hse"], c["b"], z2, C),
   578	                                     mean_share_downweighted=float(np.mean(v["hsh"]))),
   579	                    "winsorised_ols_dk": t6_block(v["wb"], v["wse"], c["b"], z2, C)}
   580	                state["cells"].append(out)
   581	                flush()
   582	                tot += n
   583	            return tot
   584	
   585	        run_task("T6", t6, 4)
   586	
   587	        # ------------------------------------------------ T3 (cells 9 to 11)
   588	        wf_calls = [0]
   589	
   590	        def t3():
   591	            pool = state["pooled"]
   592	            cb95 = pool["c_boot"]["q95"]["value"] if pool else None
   593	            cc95 = pool["c_cal"]["q95"]["value"] if pool else None
   594	            state["t3"] = {"c_cw_cal": None, "cells": [], "walk_forward_calls": 0, "registry_lines": None}
   595	            tot, ccw = 0, None
   596	            for cid in (9, 10, 11):
   597	                c = cells[cid]
   598	                n = reps(c)
   599	                v = {k: np.empty(n) for k in ("b", "se", "t", "cw", "cwp")}
   600	                for r in range(n):
   601	                    s = seed(c, r)
   602	                    y = cal.draw(c["b"], float(cfg["a2"]), seed=s, n_dates=c["n_dates"])
   603	                    y["excl_inference"] = False
   604	                    y["excl_forecast_eval"] = False
   605	                    preds, _ = F.walk_forward(y, {"M2": M2, "M3": M3}, spec, target="y_sim", seed=s)
   606	                    wf_calls[0] += 1
   607	                    state["t3"]["walk_forward_calls"] = wf_calls[0]
   608	                    cut = np.sort(y["date"].unique())[-cfg["T_conf"]]
   609	                    sub = preds[preds["date"] >= cut]
   610	                    FT.pair(sub, "M3", "M2")
   611	                    cw = FT.clark_west(sub, "M3", "M2", spec)
   612	                    assert_lag(cw["lag"], f"cell {cid} Clark-West")
   613	                    f = C.fit_t(y[y["date"] >= cut], spec, "y_sim")
   614	                    assert_lag(f["lag"], f"cell {cid} H1")
   615	                    v["b"][r], v["se"][r], v["t"][r], v["cw"][r], v["cwp"][r] = f["b"], f["se"], f["t"], cw["stat"], cw["p"]
   616	                if cid == cfg["cw_cal_cell"]:
   617	                    state["t3"]["c_cw_cal"] = os_summary(v["cw"], 0.95, C)
   618	                    ccw = state["t3"]["c_cw_cal"]["value"]
