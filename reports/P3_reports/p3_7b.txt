def defs(rel):
    """[(start, end, qualname, kind)] for every function and class (start includes decorators); P3-7a2's rule."""
    if rel in DEFS:

    out = []

    def visit(node, prefix):
        for ch in ast.iter_child_nodes(node):
            if isinstance(ch, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):

                visit(ch, q)
