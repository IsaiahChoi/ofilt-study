"""P6-1: shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p6_1_msg.txt, list
"## Expected values", one line per name, `- NAME = value` (a value keeps any leading space after "= ", as POST_STATUS's
first item does); the item's date D from /tmp/p6_1_date.out (a.1).
"""
import datetime
import hashlib
import json
import os
import re
import subprocess

MSG = "/tmp/p6_1_msg.txt"
DATE_FILE = "/tmp/p6_1_date.out"
PLACEHOLDER = re.compile(rb"\{[A-Z0-9_]+\}")
REPO = os.path.expanduser("~/Desktop/FX_Project/ofilt_pass1")
DATA = os.path.expanduser("~/Desktop/FX_Project/data")


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    print(f"D (from {DATE_FILE}): {d}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    return value.split(" ")


def barred(value):
    """Items separated by ` | ` (each kept as written, leading spaces included)."""
    return value.split(" | ")


def pairs(value):
    """`name:value` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in barred(value)]


def git_bytes(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], check=True, capture_output=True).stdout


def git(*args, repo=REPO):
    return git_bytes(*args, repo=repo).decode("utf-8")


def git_rc(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], capture_output=True).returncode


def lines_of(data):
    return data.count(b"\n")


def repo_file(rel):
    """A REPO path, refused if it would leave REPO or reach DATA (nothing in DATA is opened)."""
    full = os.path.realpath(os.path.join(REPO, rel))
    if os.path.isabs(rel) or ".." in rel.split("/") or not full.startswith(os.path.realpath(REPO) + os.sep) \
            or full.startswith(os.path.realpath(DATA) + os.sep):
        raise SystemExit(f"refused: {rel!r} is not a REPO path")
    return full


def access_log_hits(root=REPO):
    """Names starting ACCESS_LOG under root outside .git (names only), relative to root, sorted."""
    hits = []
    for r, dirs, files in os.walk(root):
        if r == root and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(r, n), root) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return sorted(hits)


def porcelain_lines(untracked="all"):
    """`git status --porcelain --untracked-files=<untracked>`'s lines, exactly as printed."""
    return [ln for ln in git("status", "--porcelain", f"--untracked-files={untracked}").split("\n") if ln]


APPX_HEAD = "## Appendix A: the files that fix the confirmation (SHA-256 at the tag)"
APPX_ROW = re.compile(r"^\| `([^`]+)` \| `([0-9a-f]{64})` \|.*\|$")
RUN_MARK = "OFILT_P6_CONFIRM_RUN=P6-1 caffeinate"


def appendix_rows_from(text):
    lines = text.split("\n")
    if lines.count(APPX_HEAD) != 1:
        raise SystemExit("the Appendix A heading does not occur exactly once")
    rows, near = [], []
    for ln in lines[lines.index(APPX_HEAD):]:
        m = APPX_ROW.match(ln)
        if m:
            rows.append((m.group(1), m.group(2)))
        elif ln.startswith("| `"):
            near.append(ln)
    if near:
        raise SystemExit(f"{len(near)} Appendix A lines start '| `' without the row form")
    return rows


def appendix_rows():
    return appendix_rows_from(read_bytes(repo_file("docs/PREREG_v1.md")).decode("utf-8"))


def appendix_check(C, E, skip=()):
    """Every Appendix A file (except those in skip) has the row's SHA-256 in the working tree."""
    rows = appendix_rows()
    C.check(len(rows) == int(E["APPX_ROWS"]) and len({p for p, _ in rows}) == len(rows),
            "Appendix A: APPX_ROWS rows, no path twice", f"{len(rows)}")
    todo = [(p, h) for p, h in rows if p not in skip]
    bad = [p for p, h in todo if sha256(repo_file(p)) != h]
    C.check(not bad, f"every Appendix A file{' other than ' + ', '.join(skip) if skip else ''} has the row's sha256 "
            f"(working tree)", f"{len(todo) - len(bad)} of {len(todo)}" + ("; mismatch: " + " ".join(bad) if bad else ""))
    return rows

def sanction_check(C, E, data, label):
    """a.3's checks of a sanction text (bytes)."""
    print(f"----- {label}, in full -----")
    print(data.decode("utf-8", errors="replace"), end="")
    print(f"----- end of {label} -----")
    print(f"{label} sha256 {sha256_bytes(data)}")
    try:
        text = data.decode("utf-8")
        C.check(True, f"{label} is UTF-8")
    except UnicodeDecodeError:
        C.check(False, f"{label} is UTF-8")
        return
    C.check(text.endswith("\n"), f"{label} ends in a newline")
    lines = text.split("\n")
    C.check(len(lines) == 10 and lines[-1] == "", f"{label} has exactly nine lines, nothing after the ninth",
            f"{len(lines) - 1} newline-ended lines; trailing {lines[-1]!r}")
    if len(lines) < 10:
        return
    C.check(re.fullmatch(r"Date: [0-9]{4}-[0-9]{2}-[0-9]{2}", lines[1]) is not None,
            "line 2 matches ^Date: YYYY-MM-DD$", lines[1])
    try:
        datetime.date.fromisoformat(lines[1][len("Date: "):])
        C.check(True, "line 2 is a valid calendar date", lines[1][len("Date: "):])
    except ValueError:
        C.check(False, "line 2 is a valid calendar date", lines[1])
    for i in (1, 3, 4, 5, 6, 7, 8, 9):
        C.check(lines[i - 1] == E[f"SANCTION_LINE{i}"], f"line {i} == SANCTION_LINE{i}")
    C.check(lines[3] == f"HEAD: {E['S']}", "line 4's commit is S", lines[3])
    for i, (p, h) in zip((5, 6, 7), pairs(E["SANCTIONED"])):
        C.check(lines[i - 1] == f"{p}: {h}", f"line {i} holds SANCTIONED's hash for {p}")
    m = re.fullmatch(r"Criteria: docs/P6_RUN_CRITERIA\.md ([0-9a-f]{64})", lines[7])
    named = m.group(1) if m else None
    C.check(named == E["CRITERIA_SHA256"], "line 8's hash is CRITERIA_SHA256", str(named))
    at_s = sha256_bytes(git_bytes("show", f"{E['S']}:docs/P6_RUN_CRITERIA.md"))
    C.check(named == at_s, "line 8's hash == sha256 of git show S:docs/P6_RUN_CRITERIA.md", at_s)
    status_s = git("show", f"{E['S']}:PASS6_STATUS.md")
    C.check(named is not None and named in status_s, "git show S:PASS6_STATUS.md contains it")


def verdict_of(out_path, name):
    """True iff the file holds exactly one line `<name> CHECK: True` and no `<name> CHECK: False`."""
    if not os.path.exists(out_path):
        return False
    lines = read_bytes(out_path).decode("utf-8").split("\n")
    return lines.count(f"{name} CHECK: True") == 1 and f"{name} CHECK: False" not in lines


def substitute(template_bytes, d, p6_0_sha, sanction_sha):
    return (template_bytes.replace(b"{DATE}", d.encode()).replace(b"{P6_0_SHA256}", p6_0_sha.encode())
            .replace(b"{SANCTION_SHA256}", sanction_sha.encode()))


def counts_check(C, label, data, counts):
    """counts: `{NAME}:n | ...`; exactly those placeholders, each n times, no other."""
    want = {k.encode(): int(v) for k, v in pairs(counts)}
    for k, n in want.items():
        C.check(data.count(k) == n, f"{label} holds {k.decode()} {n} times", f"{data.count(k)}")
    other = [m for m in PLACEHOLDER.findall(data) if m not in want]
    C.check(not other, f"{label} holds no other placeholder", b" ".join(other).decode())


def placeholder_lines(template_bytes, written_bytes):
    tl, wl = template_bytes.decode("utf-8").split("\n"), written_bytes.decode("utf-8").split("\n")
    print("each template line that held a placeholder, as written:")
    for i, (a, b) in enumerate(zip(tl, wl), 1):
        if PLACEHOLDER.search(a.encode("utf-8")):
            print(f"  {i}: {b}")
    return len(tl) == len(wl)


def strict_json(data):
    def refuse(c):
        raise ValueError(f"non-finite constant {c}")
    return json.loads(data.decode("utf-8"), parse_constant=refuse)


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def note(self, text):
        print(f"[note] {text}")

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")
        return not self.fails

    """Strict JSON of a REPO file, or None (FAIL recorded) if it is absent or does not parse."""

        return None

# P6-1 g.1: the bases still have TESTKIT_BASE and tests/run_state.py is absent; then cp -p the four kit files from

MSG="/tmp/p6_1_msg.txt"
echo "message $MSG: sha256 $(shasum -a 256 "$MSG" | cut -d' ' -f1)"
ev() {
  local n
  n="$(grep -c -E "^- $1 = " "$MSG")"
  [ "$n" = "1" ] || { echo "REFUSED: expected value $1 found $n times"; exit 2; }
  grep -E "^- $1 = " "$MSG" | sed -E "s/^- $1 = //"
}
KIT="$(ev KIT_DIR)"; KIT="${KIT/#\~/$HOME}/tests"
splitbar() {  # items separated by " | ", one per line
  local rest="$1" item
  while :; do item="${rest%% | *}"; printf '%s\n' "$item"; [ "$item" = "$rest" ] && break; rest="${rest#* | }"; done
}
while IFS= read -r item; do
  p="${item%:*}"; want="${item##*:}"
  got="$(shasum -a 256 "$REPO/$p" | cut -d' ' -f1)"
  if [ "$got" = "$want" ]; then echo "base $p: $got (TESTKIT_BASE)"; else echo "REFUSED: base $p is $got, not $want"; exit 2; fi
done < <(splitbar "$(ev TESTKIT_BASE)")
[ ! -e "$REPO/tests/run_state.py" ] || { echo "REFUSED: tests/run_state.py exists"; exit 2; }
echo "tests/run_state.py: absent"
n=0
while IFS= read -r item; do
  f="${item%:*}"
  cp -p "$KIT/$f" "$REPO/tests/$f"
  echo "cp -p $KIT/$f -> tests/$f"
  n=$((n + 1))
done < <(splitbar "$(ev TESTKIT)")
echo "installed: $n"
