"""P6-1c: shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p6_1c_msg.txt, list

import hashlib

def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8")
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}\n", d):
        raise SystemExit(f"{DATE_FILE} does not hold one date line: {d!r}")
    print(f"D (from {DATE_FILE}): {d.strip()}")
    return d.strip()


def r_commit():

    lines = [ln for ln in read_bytes("/tmp/p6_1c_b_commit.out").decode("utf-8").split("\n")

    R = lines[0].split()[1]
    print(f"R (from /tmp/p6_1c_b_commit.out): {R}")

def path(value):
    return os.path.expanduser(value)


def spaced(value):
    return value.split(" ")


def barred(value):
    """Items separated by ` | ` (each kept as written, leading spaces included)."""
    return value.split(" | ")


def pairs(value):
    return [tuple(item.rsplit(":", 1)) for item in barred(value)]


def git_bytes(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], check=True, capture_output=True).stdout


def git(*args, repo=REPO):
    return git_bytes(*args, repo=repo).decode("utf-8")


def git_rc(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], capture_output=True).returncode


def lines_of(data):
    return data.count(b"\n")


def repo_file(rel):
    """A REPO path, refused if it would leave REPO or reach DATA (nothing in DATA is opened)."""
    full = os.path.realpath(os.path.join(REPO, rel))
    if os.path.isabs(rel) or ".." in rel.split("/") or not full.startswith(os.path.realpath(REPO) + os.sep) \
            or full.startswith(os.path.realpath(DATA) + os.sep):
        raise SystemExit(f"refused: {rel!r} is not a REPO path")
    return full


def regular(p):
    return os.path.isfile(p) and not os.path.islink(p)


def access_log_hits(root=REPO):
    """Names starting ACCESS_LOG under root outside .git (names only), relative to root, sorted."""
    hits = []
    for r, dirs, files in os.walk(root):
        if r == root and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(r, n), root) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return sorted(hits)


def porcelain_lines(untracked="all"):
    return [ln for ln in git("status", "--porcelain", f"--untracked-files={untracked}").split("\n") if ln]


def name_status(*args):
    """git name-status lines as `<status> <path>[ <path>]` (tabs replaced by single spaces)."""
    return [ln.replace("\t", " ") for ln in git(*args).split("\n") if ln]


APPX_HEAD = "## Appendix A: the files that fix the confirmation (SHA-256 at the tag)"
APPX_ROW = re.compile(r"^\| `([^`]+)` \| `([0-9a-f]{64})` \|.*\|$")


def appendix_rows():
    lines = read_bytes(repo_file("docs/PREREG_v1.md")).decode("utf-8").split("\n")
    if lines.count(APPX_HEAD) != 1:
        raise SystemExit("the Appendix A heading does not occur exactly once")

    for ln in lines[lines.index(APPX_HEAD):]:
        m = APPX_ROW.match(ln)

        elif ln.startswith("| `"):
            near.append(ln)
    if near:
        raise SystemExit(f"{len(near)} Appendix A lines start '| `' without the row form")
    return rows

def counts_check(C, label, data, counts):
    """counts: `{NAME}:n | ...`; exactly those placeholders, each n times, no other."""
    want = {k.encode(): int(v) for k, v in pairs(counts)}
    for k, n in want.items():
        C.check(data.count(k) == n, f"{label} holds {k.decode()} {n} times", f"{data.count(k)}")
    other = [m for m in PLACEHOLDER.findall(data) if m not in want]
    C.check(not other, f"{label} holds no other placeholder", b" ".join(other).decode())


def substitute(template_bytes, subs):
    out = template_bytes
    for k, v in subs.items():
        out = out.replace(k.encode("utf-8"), v.encode("utf-8"))
    return out

def placeholder_lines(template_bytes, written_bytes):
    tl, wl = template_bytes.decode("utf-8").split("\n"), written_bytes.decode("utf-8").split("\n")
    print("each template line that held a placeholder, as written:")
    for i, (a, b) in enumerate(zip(tl, wl), 1):
        if PLACEHOLDER.search(a.encode("utf-8")):
            print(f"  {i}: {b}")
    return len(tl) == len(wl)


def unittest_out_check(C, out_path, ran_n, result):
    out = read_bytes(out_path).decode("utf-8").split("\n")
    ran = [ln for ln in out if re.fullmatch(r"Ran \d+ tests? in [0-9.]+s", ln)]
    C.check(len(ran) == 1 and ran[0].startswith(f"Ran {ran_n} tests "), f"{out_path}: Ran {ran_n} tests", " | ".join(ran))
    C.check(out.count(result) == 1, f"{out_path}: a line exactly {result}")
    bad = [ln for ln in out if ln.startswith("FAIL:") or ln.startswith("ERROR:")]
    C.check(not bad, f"{out_path}: no FAIL: or ERROR: line", " | ".join(bad))


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def note(self, text):
        print(f"[note] {text}")

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")
        return not self.fails
