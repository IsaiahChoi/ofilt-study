"""P6-boot: shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p6_boot_msg.txt, list
"## Expected values", one line per name, `- NAME = value`; the item's date D from /tmp/p6_boot_date.out (a.1).
"""
import hashlib
import os
import re
import subprocess

MSG = "/tmp/p6_boot_msg.txt"
DATE_FILE = "/tmp/p6_boot_date.out"


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    print(f"D (from {DATE_FILE}): {d}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    """Items separated by a space."""
    return value.split(" ")


def pairs(value):
    """`path:sha256` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in value.split(" | ")]


REPO = path("~/Desktop/FX_Project/ofilt_pass1")
DATA = path("~/Desktop/FX_Project/data")


def git_bytes(*args, repo=REPO, check=True):
    return subprocess.run(["git", "-C", repo, *args], check=check, capture_output=True).stdout


def git(*args, repo=REPO):
    return git_bytes(*args, repo=repo).decode("utf-8")


def git_rc(*args, repo=REPO):
    """Exit status only."""
    return subprocess.run(["git", "-C", repo, *args], capture_output=True).returncode


def lines_of(data):
    return data.count(b"\n")


def repo_file(rel):
    """A REPO path, refused if it would leave REPO or reach DATA (nothing in DATA is opened)."""
    full = os.path.realpath(os.path.join(REPO, rel))
    if os.path.isabs(rel) or ".." in rel.split("/") or not full.startswith(os.path.realpath(REPO) + os.sep) \
            or full.startswith(os.path.realpath(DATA) + os.sep):
        raise SystemExit(f"refused: {rel!r} is not a REPO path")
    return full


def access_log_hits(root=REPO):
    """Names starting ACCESS_LOG under root outside .git (names only; nothing opened, other names not printed)."""
    hits = []
    for r, dirs, files in os.walk(root):
        if r == root and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(r, n), root) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return hits


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def note(self, text):
        print(f"[note] {text}")

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")

"""P6-boot a.5: REPO's tracked files against the H11 manifest; every file hashed as bytes (nothing printed of any

C.check(len(lines) == int(E["MANIFEST_LINES"]), "manifest lines == MANIFEST_LINES", f"{len(lines)}")
man = {}
bad_form = 0
dups = []
for ln in lines:
    m = re.fullmatch(r"([0-9a-f]{64})\t([^\t]+)", ln)
    if not m:
        bad_form += 1
        continue
    if m.group(2) in man:
        dups.append(m.group(2))
    man[m.group(2)] = m.group(1)
C.check(bad_form == 0, "every manifest line is <sha256><TAB><path>", f"{bad_form} lines of another form")
C.check(not dups, "no manifest path twice", " ".join(dups))

tracked = [p.decode("utf-8") for p in git_bytes("ls-files", "-z").split(b"\0") if p]
C.check(len(tracked) == len(set(tracked)), "git ls-files -z lists no path twice")
C.check(len(tracked) == int(E["TRACKED_COUNT"]), "tracked files == TRACKED_COUNT", f"{len(tracked)}")
only_git = sorted(set(tracked) - set(man))
only_man = sorted(set(man) - set(tracked))
C.check(not only_git and not only_man, "git ls-files -z lists exactly the manifest's paths",
        f"tracked only: {len(only_git)}; manifest only: {len(only_man)}")
for p in only_git:
    print(f"  tracked, not in the manifest: {p}")
for p in only_man:
    print(f"  in the manifest, not tracked: {p}")

links = [p for p in tracked if os.path.islink(os.path.join(os.path.expanduser("~/Desktop/FX_Project/ofilt_pass1"), p))]
C.check(not links, "no tracked path is a symlink", " ".join(links))

matched = 0
mismatched = []
for p in sorted(set(tracked) & set(man)):
    if sha256(repo_file(p)) == man[p]:
        matched += 1
    else:
        mismatched.append(p)
print(f"hashed: {matched + len(mismatched)}; matching the manifest: {matched}; mismatched: {len(mismatched)}")
for p in mismatched:
    print(f"  mismatch: {p}")
C.check(not mismatched and matched == int(E["TRACKED_COUNT"]),
        "every tracked file's working-tree bytes have the manifest's sha256", f"{matched} of {len(tracked)}")

C.done()

def at_tag(rel):
    shown = git_bytes("show", f"prereg-v1:{rel}")
    blob = git_bytes("cat-file", "blob", f"prereg-v1:{rel}")
    if shown != blob:
        raise SystemExit(f"git show and git cat-file blob differ for {rel}")
    return shown


ok_repo = ok_tag = 0
for rel, want in rows:
    h_repo = sha256(repo_file(rel))
    h_tag = sha256_bytes(at_tag(rel))
    ok_repo += h_repo == want
    ok_tag += h_tag == want
    print(f"  {'ok' if h_repo == want else 'MISMATCH'} in REPO, {'ok' if h_tag == want else 'MISMATCH'} at the tag: {rel}")
C.check(ok_repo == len(rows), "every row's file in REPO has the row's sha256", f"{ok_repo} of {len(rows)}")

    hits = [ln for ln in text.split("\n") if CMD_MARK in ln]

    return out
