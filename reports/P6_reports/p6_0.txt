"""P6-0: shared helpers for the item's Python scripts.

Imports no module of ofilt; reads files as bytes. Expected values are read from /tmp/p6_0_msg.txt, list
"## Expected values", one line per name, `- NAME = value`; the item's date D from /tmp/p6_0_date.out (a.1).
"""
import hashlib
import os
import re
import subprocess

MSG = "/tmp/p6_0_msg.txt"
DATE_FILE = "/tmp/p6_0_date.out"
PLACEHOLDER = re.compile(rb"\{[A-Z0-9_]+\}")


def read_bytes(path):
    with open(path, "rb") as fh:
        return fh.read()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256(path):
    return sha256_bytes(read_bytes(path))


def expected():
    data = read_bytes(MSG)
    print(f"message {MSG}: sha256 {sha256_bytes(data)}")
    lines = data.decode("utf-8").split("\n")
    start = lines.index("## Expected values")
    E = {}
    for ln in lines[start + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            if m.group(1) in E:
                raise SystemExit(f"duplicate expected value {m.group(1)}")
            E[m.group(1)] = m.group(2)
    print(f"expected values read: {len(E)}")
    return E


def item_date():
    d = read_bytes(DATE_FILE).decode("utf-8").strip()
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", d):
        raise SystemExit(f"{DATE_FILE} does not hold a date: {d!r}")
    print(f"D (from {DATE_FILE}): {d}")
    return d


def path(value):
    return os.path.expanduser(value)


def spaced(value):
    """Items separated by a space."""
    return value.split(" ")


def pairs(value):
    """`name:sha256` items separated by ` | `, split at the last colon."""
    return [tuple(item.rsplit(":", 1)) for item in value.split(" | ")]


def arrows(value, E):
    """`source>repo path` items separated by ` | `; a source that is an Expected-value key is replaced by its value;
    `~` expanded."""
    out = []
    for item in value.split(" | "):
        src, dst = item.split(">")
        out.append((path(E[src] if src in E else src), dst))
    return out


REPO = path("~/Desktop/FX_Project/ofilt_pass1")
DATA = path("~/Desktop/FX_Project/data")


def git_bytes(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], check=True, capture_output=True).stdout


def git(*args, repo=REPO):
    return git_bytes(*args, repo=repo).decode("utf-8")


def git_rc(*args, repo=REPO):
    return subprocess.run(["git", "-C", repo, *args], capture_output=True).returncode


def lines_of(data):
    return data.count(b"\n")


def repo_file(rel):
    """A REPO path, refused if it would leave REPO or reach DATA (nothing in DATA is opened)."""
    full = os.path.realpath(os.path.join(REPO, rel))
    if os.path.isabs(rel) or ".." in rel.split("/") or not full.startswith(os.path.realpath(REPO) + os.sep) \
            or full.startswith(os.path.realpath(DATA) + os.sep):
        raise SystemExit(f"refused: {rel!r} is not a REPO path")
    return full


def access_log_hits(root=REPO):
    """Names starting ACCESS_LOG under root outside .git (names only; nothing opened, other names not printed)."""
    hits = []
    for r, dirs, files in os.walk(root):
        if r == root and ".git" in dirs:
            dirs.remove(".git")
        hits += [os.path.relpath(os.path.join(r, n), root) for n in dirs + files if n.startswith("ACCESS_LOG")]
    return hits


def porcelain():
    """`git status --porcelain --untracked-files=all` as a list of (status, path)."""
    out = git("status", "--porcelain", "--untracked-files=all")
    return [(ln[:2], ln[3:]) for ln in out.splitlines() if ln]


APPX_HEAD = "## Appendix A: the files that fix the confirmation (SHA-256 at the tag)"
APPX_ROW = re.compile(r"^\| `([^`]+)` \| `([0-9a-f]{64})` \|.*\|$")


def appendix_rows():
    """Appendix A's (path, sha256) rows, parsed from docs/PREREG_v1.md's working-tree bytes (as P6-boot a.6)."""
    lines = read_bytes(repo_file("docs/PREREG_v1.md")).decode("utf-8").split("\n")
    if lines.count(APPX_HEAD) != 1:
        raise SystemExit("the Appendix A heading does not occur exactly once")
    rows, near = [], []
    for ln in lines[lines.index(APPX_HEAD):]:
        m = APPX_ROW.match(ln)
        if m:
            rows.append((m.group(1), m.group(2)))
        elif ln.startswith("| `"):
            near.append(ln)
    if near:
        raise SystemExit(f"{len(near)} Appendix A lines start '| `' without the row form")
    return rows


def appendix_check(C, E):
    rows = appendix_rows()
    C.check(len(rows) == int(E["APPX_ROWS"]) and len({p for p, _ in rows}) == len(rows),
            "Appendix A: APPX_ROWS rows, no path twice", f"{len(rows)}")
    bad = [p for p, h in rows if sha256(repo_file(p)) != h]
    C.check(not bad, "every Appendix A file's working-tree bytes have the row's sha256",
            f"{len(rows) - len(bad)} of {len(rows)}" + ("; mismatch: " + " ".join(bad) if bad else ""))


def logs_check(C, E):

        data = read_bytes(repo_file(rel))

                f"{rel} sha256 == {key}_SHA256 and lines == {key}_LINES", f"{sha256_bytes(data)}, {lines_of(data)} lines")


def tags_check(C, E):
    tags = git("tag", "-l").split()
    C.check(tags == spaced(E["TAG_LIST"]), "git tag -l == TAG_LIST", " ".join(tags))
    kind = git("cat-file", "-t", "refs/tags/prereg-v1").strip()

    commit = git("rev-parse", "refs/tags/prereg-v1^{commit}").strip()
    C.check(commit == E["PREREG_TAG_COMMIT"], "prereg-v1's commit == PREREG_TAG_COMMIT", commit)


def access_check(C):

    C.check(hits == [], "no name starting ACCESS_LOG in REPO outside .git (names only)", " ".join(hits))


def status_template(E, d, boot_sha):
    """(template bytes, PASS6_STATUS.md bytes): every {DATE} -> D, every {P6_BOOT_SHA256} -> P6-boot's report's sha."""
    t = read_bytes(os.path.join(path(E["KIT_DIR"]), "PASS6_STATUS.template.md"))
    return t, t.replace(b"{DATE}", d.encode("utf-8")).replace(b"{P6_BOOT_SHA256}", boot_sha.encode("utf-8"))


class Checks:
    def __init__(self, verdict):
        self.verdict = verdict
        self.n = 0
        self.fails = []

    def check(self, ok, label, detail=""):
        self.n += 1
        if not ok:
            self.fails.append(label)
        print(f"[{'ok' if ok else 'FAIL'}] {label}" + (f": {detail}" if detail else ""))

    def note(self, text):
        print(f"[note] {text}")

    def done(self):
        print(f"checks: {self.n}; failed: {len(self.fails)}")
        for f in self.fails:
            print(f"FAIL {f}")
        print(f"{self.verdict} CHECK: {not self.fails}")

#!/bin/bash
# P6-0 b.1: installs INSTALL's three files with cp -p. INSTALL and the keyed sources (P5_4_KEPT, P6_BOOT_KEPT) are read
# from the message's Expected values by script. Every destination is checked absent before any copy. Runs no Python.

REPO="$HOME/Desktop/FX_Project/ofilt_pass1"
MSG="/tmp/p6_0_msg.txt"
echo "message $MSG: sha256 $(shasum -a 256 "$MSG" | cut -d' ' -f1)"
ev() {  # the value of one expected value, exactly once
  local n
  n="$(grep -c -E "^- $1 = " "$MSG")"
  [ "$n" = "1" ] || { echo "REFUSED: expected value $1 found $n times"; exit 2; }
  grep -E "^- $1 = " "$MSG" | sed -E "s/^- $1 = //"
}
INSTALL="$(ev INSTALL)"
echo "INSTALL = $INSTALL"
SRCS=(); DSTS=()
rest="$INSTALL"
while :; do  # items separated by " | "
  item="${rest%% | *}"
  src="${item%%>*}"; dst="${item#*>}"
  if [[ "$src" =~ ^[A-Z0-9_]+$ ]]; then src="$(ev "$src")"; fi
  src="${src/#\~/$HOME}"
  SRCS+=("$src"); DSTS+=("$dst")
  [ "$item" = "$rest" ] && break
  rest="${rest#* | }"
done
[ "${#SRCS[@]}" -eq 3 ] || { echo "REFUSED: INSTALL has ${#SRCS[@]} items, not 3"; exit 2; }
for i in "${!SRCS[@]}"; do
  [ -f "${SRCS[$i]}" ] || { echo "REFUSED: source ${SRCS[$i]} is not a file"; exit 2; }
  [ ! -e "$REPO/${DSTS[$i]}" ] || { echo "REFUSED: destination ${DSTS[$i]} exists"; exit 2; }
done
for i in "${!SRCS[@]}"; do
  mkdir -p "$(dirname "$REPO/${DSTS[$i]}")"
  cp -p "${SRCS[$i]}" "$REPO/${DSTS[$i]}"
  echo "cp -p ${SRCS[$i]} -> ${DSTS[$i]}"
done
echo "installed: ${#SRCS[@]}"
