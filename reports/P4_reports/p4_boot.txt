# P4-boot shared helpers: every expected value is read from /tmp/p4_boot_msg.txt, never typed.
import hashlib
import os
import re
import subprocess
from pathlib import Path

HOME = os.environ["HOME"]
MSG = Path("/tmp/p4_boot_msg.txt").read_text()
REPO = Path(HOME) / "Desktop/FX_Project/ofilt_pass1"
FILES = Path(HOME) / "Desktop/FX_Project/files"


def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def git(*args):
    return subprocess.run(["git", "-C", str(REPO), *args],
                          capture_output=True, text=True, check=True).stdout


def one(pattern):
    found = re.findall(pattern, MSG)
    assert len(found) == 1, (pattern, found)
    return found[0]


def table(header):
    """Rows of the markdown table whose header line is exactly `header`."""
    lines = MSG.splitlines()
    assert lines.count(header) == 1, header
    i = lines.index(header)
    assert set(lines[i + 1].replace("|", "")) == {"-"}, lines[i + 1]
    rows = []
    for ln in lines[i + 2:]:
        if not ln.startswith("|"):
            break
        cells = [c.strip() for c in ln.strip().strip("|").split("|")]
        assert len(cells) == 2 and all(c.startswith("`") and c.endswith("`") for c in cells), ln
        rows.append((cells[0][1:-1], cells[1][1:-1]))
    return rows


def expand(p):
    return p.replace("$HOME", HOME)


def show(label, expected, actual):
    ok = expected == actual
    print(f"{label}: expected={expected!r} actual={actual!r} ok={ok}")
    return ok
