# P4-4a shared helpers. Every expected value is read from /tmp/p4_4a_msg.txt's list "Expected values" by its label.
# Reads files as text or bytes; imports no module of ofilt.
import hashlib
import os
import subprocess
from pathlib import Path

HOME = os.environ["HOME"]
MSG = Path("/tmp/p4_4a_msg.txt").read_text(encoding="utf-8")
FXP = Path(HOME) / "Desktop/FX_Project"
REPO = FXP / "ofilt_pass1"
PROMPT = FXP / "files/pass4/PASS4_BUILDER_PROMPT_2026-10-08.md"
KIT = FXP / "files/pass4/p4_4a_kit"
KEEP4 = FXP / "files/pass4_tmp_keep"
_SEC = MSG.split("\n## Expected values\n")[1].split("\n## ")[0].split("\n")


def ev(name):
    hits = [ln for ln in _SEC if ln.startswith(f"- {name} = ")]
    assert len(hits) == 1, name
    return hits[0][len(f"- {name} = "):]


def evlist(name):
    return ev(name).split(" ")


def sha256_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha256(p):
    return sha256_bytes(Path(p).read_bytes())


def git(*args, check=True):
    return subprocess.run(["git", "-C", str(REPO), *args], capture_output=True, text=True, check=check)


def show(label, expected, actual):
    ok = expected == actual
    print(f"{label}: expected={expected!r} actual={actual!r} ok={ok}")
    return ok


def status_lines():
    return git("status", "--porcelain", "--untracked-files=all").stdout.splitlines()


def tripwires(oks):
    """Tags, BLIND_FIT_LOG, RUN_REGISTRY.jsonl, ACCESS_LOG and NODATA, as in step a.2 (the tree is checked by each step)."""
    oks.append(show("git tag -l", evlist("TAG_LIST"), git("tag", "-l").stdout.split()))
    oks.append(show("refs/tags/spec-dev-v1", ev("TAG_OBJECT"), git("rev-parse", "refs/tags/spec-dev-v1").stdout.strip()))
    oks.append(show("spec-dev-v1^{commit}", ev("TAG_COMMIT"), git("rev-parse", "refs/tags/spec-dev-v1^{commit}").stdout.strip()))
    bfl = (REPO / "BLIND_FIT_LOG").read_bytes()
    oks.append(show("BLIND_FIT_LOG sha256", ev("BLIND_FIT_LOG_SHA256"), sha256_bytes(bfl)))
    oks.append(show("BLIND_FIT_LOG lines", int(ev("BLIND_FIT_LOG_LINES")), bfl.count(b"\n")))
    oks.append(show("RUN_REGISTRY.jsonl bytes", int(ev("RUN_REGISTRY_BYTES")), (REPO / "RUN_REGISTRY.jsonl").stat().st_size))
    found = []
    for d, dirs, files in os.walk(REPO):
        dirs[:] = [x for x in dirs if not (d == str(REPO) and x == ".git")]
        found += [os.path.join(d, x) for x in dirs + files if x.startswith("ACCESS_LOG")]
    oks.append(show("names starting ACCESS_LOG in REPO outside .git", [], found))
    nd = Path(ev("NODATA"))
    oks.append(show("NODATA exists and is empty", (True, []), (nd.is_dir(), sorted(os.listdir(nd)) if nd.is_dir() else None)))


def pinned(oks, labels=("PROMPT_SHA256", "SPEC_SHA256", "PASS3_STATUS_SHA256", "CONFIG_SHA256", "SIM_MODULE_SHA256",
                        "SIM_TESTS_SHA256", "SIM_SPEC_SHA256")):

             "SIM_TESTS_SHA256": REPO / "tests/test_p4_sim_full.py", "SIM_SPEC_SHA256": REPO / "docs/P4_SIM_SPEC.md"}
    for label in labels:
        oks.append(show(f"sha256 {paths[label].name} ({label})", ev(label), sha256(paths[label])))

# P4-4a (message version 2) shared helpers. Every expected value is read from /tmp/p4_4a_msg.txt's list
# "Expected values" by its label. Reads files as text or bytes; imports no module of ofilt.
import hashlib
import os
import subprocess
from pathlib import Path

HOME = os.environ["HOME"]
MSG = Path("/tmp/p4_4a_msg.txt").read_text(encoding="utf-8")
FXP = Path(HOME) / "Desktop/FX_Project"
REPO = FXP / "ofilt_pass1"
PROMPT = FXP / "files/pass4/PASS4_BUILDER_PROMPT_2026-10-08.md"
KIT = FXP / "files/pass4/p4_4av2_kit"
KEEP4 = FXP / "files/pass4_tmp_keep"
_SEC = MSG.split("\n## Expected values\n")[1].split("\n## ")[0].split("\n")


def ev(name):
    hits = [ln for ln in _SEC if ln.startswith(f"- {name} = ")]
    assert len(hits) == 1, name
    return hits[0][len(f"- {name} = "):]


def evlist(name):
    return ev(name).split(" ")


def sha256_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha256(p):
    return sha256_bytes(Path(p).read_bytes())


def git(*args, check=True):
    return subprocess.run(["git", "-C", str(REPO), *args], capture_output=True, text=True, check=check)


def show(label, expected, actual):
    ok = expected == actual
    print(f"{label}: expected={expected!r} actual={actual!r} ok={ok}")
    return ok


def status_lines():
    return git("status", "--porcelain", "--untracked-files=all").stdout.splitlines()


def tripwires(oks):
    """Tags, BLIND_FIT_LOG, RUN_REGISTRY.jsonl, ACCESS_LOG and NODATA, as in step a.2 (the tree is checked by each step)."""
    oks.append(show("git tag -l", evlist("TAG_LIST"), git("tag", "-l").stdout.split()))
    oks.append(show("refs/tags/spec-dev-v1", ev("TAG_OBJECT"), git("rev-parse", "refs/tags/spec-dev-v1").stdout.strip()))
    oks.append(show("spec-dev-v1^{commit}", ev("TAG_COMMIT"), git("rev-parse", "refs/tags/spec-dev-v1^{commit}").stdout.strip()))
    bfl = (REPO / "BLIND_FIT_LOG").read_bytes()
    oks.append(show("BLIND_FIT_LOG sha256", ev("BLIND_FIT_LOG_SHA256"), sha256_bytes(bfl)))
    oks.append(show("BLIND_FIT_LOG lines", int(ev("BLIND_FIT_LOG_LINES")), bfl.count(b"\n")))
    oks.append(show("RUN_REGISTRY.jsonl bytes", int(ev("RUN_REGISTRY_BYTES")), (REPO / "RUN_REGISTRY.jsonl").stat().st_size))
    found = []
    for d, dirs, files in os.walk(REPO):
        dirs[:] = [x for x in dirs if not (d == str(REPO) and x == ".git")]
        found += [os.path.join(d, x) for x in dirs + files if x.startswith("ACCESS_LOG")]
    oks.append(show("names starting ACCESS_LOG in REPO outside .git", [], found))
    nd = Path(ev("NODATA"))
    oks.append(show("NODATA exists and is empty", (True, []), (nd.is_dir(), sorted(os.listdir(nd)) if nd.is_dir() else None)))


PINNED = {"PROMPT_SHA256": PROMPT, "SPEC_SHA256": REPO / "spec/analysis_spec_dev_v1.yaml",
          "PASS3_STATUS_SHA256": REPO / "PASS3_STATUS.md", "STATUS_BEFORE_SHA256": REPO / "PASS4_STATUS.md",
          "CONFIG_SHA256": REPO / "docs/P4_CONFIRM_CONFIG.json", "SIM_MODULE_SHA256": REPO / "ofilt/drift/sim_full.py",
          "SIM_TESTS_SHA256": REPO / "tests/test_p4_sim_full.py", "SIM_SPEC_SHA256": REPO / "docs/P4_SIM_SPEC.md"}


def pinned(oks, labels):
    for label in labels:
        oks.append(show(f"sha256 {PINNED[label].name} ({label})", ev(label), sha256(PINNED[label])))

# P4-4a d.5 and e.1 helper: read the kit's edits file (version 2) by the rules in its header, after checking its
# SHA-256, and apply a list of its edits to a text, with the placeholder kept (date None) or replaced by a date.
import re
import sys

sys.path.insert(0, "/tmp")
from p4_4a_common import KIT, PROMPT, ev, sha256  # noqa: E402

EDITS_FILE = KIT / "p4_4av2_STATUS_EDITS.txt"
MARK = re.compile(r"^@@@ (EDIT \d\d|FINAL) (OLD|NEW|END)$")


def read_edits():
    """(placeholder, {name: (old, new)}) in file order."""
    assert sha256(EDITS_FILE) == ev("KIT_EDITS_SHA256"), "edits file SHA-256"
    lines = EDITS_FILE.read_bytes().decode("utf-8").split("\n")
    if lines and lines[-1] == "":
        lines.pop()
    k = 0
    header = []
    while not MARK.match(lines[k]):
        header.append(lines[k])
        k += 1
    ph = re.findall(r"each occurrence of the placeholder (@@[A-Z]+@@) is replaced", " ".join(header))
    assert len(ph) == 1, ph
    edits = {}
    while k < len(lines):
        m = MARK.match(lines[k])
        assert m and m.group(2) == "OLD", (k, lines[k])
        name = m.group(1)
        assert name not in edits, name
        j = k + 1
        while not MARK.match(lines[j]):
            j += 1
        assert MARK.match(lines[j]).groups() == (name, "NEW"), (j, lines[j])
        e = j + 1
        while not MARK.match(lines[e]):
            e += 1
        assert MARK.match(lines[e]).groups() == (name, "END"), (e, lines[e])
        edits[name] = ("\n".join(lines[k + 1:j]), "\n".join(lines[j + 1:e]))
        k = e + 1
    return ph[0], edits


def apply(text, names, date):
    """Apply the named edits in order (date None: the placeholder kept). Returns (text or None, log lines)."""
    ph, edits = read_edits()
    log = []
    for name in names:
        old, new = edits[name]
        if date is not None:
            new = new.replace(ph, date)
        n = text.count(old)
        log.append(f"{name}: OLD occurs {n} time(s)")
        if n != 1:
            return None, log
        text = text.replace(old, new, 1)
    return text, log


def headings():
    sec = re.search(r"\*\*PASS4_STATUS\.md\*\* has these sections: (.+?)\. After", PROMPT.read_text(encoding="utf-8"))
    return ["## " + s.strip() for s in sec.group(1).split(",")]

def _iso(x) -> str:
    return pd.Timestamp(x).strftime("%Y-%m-%d")


def _plain(x: Any) -> Any:
    """The plain-typed form of section 4.3: str, int, float, bool, None, list or dict; dates as YYYY-MM-DD."""
    if x is None or isinstance(x, str):
        return x
    if x is pd.NaT:
        return None
    if isinstance(x, (bool, np.bool_)):
        return bool(x)
    if isinstance(x, (int, np.integer)):
        return int(x)
    if isinstance(x, (float, np.floating)):
        return float(x)
    if isinstance(x, dict):
        return {str(k): _plain(v) for k, v in x.items()}
    if isinstance(x, (list, tuple, np.ndarray)):
        return [_plain(v) for v in x]
    if isinstance(x, (pd.Timestamp, np.datetime64)):
        return None if pd.isna(x) else _iso(x)
    raise TypeError(f"no plain form for {type(x).__name__}")


def _all_finite(x: Any) -> bool:
    if isinstance(x, float):
        return math.isfinite(x)
    if isinstance(x, dict):
        return all(_all_finite(v) for v in x.values())
    if isinstance(x, list):
        return all(_all_finite(v) for v in x)
    return True

class _NotEstimable(Exception):

def _row(rid: str, compute: Callable[[], _Out]) -> dict:
    """The one helper of section 5: computes a row's entry and applies the contingencies."""
    try:
        out = compute()
    except _NotEstimable as e:
        out = _Out("not_estimable", None, e.reason, e.n_rows, e.n_dates, None)
    except inference.DesignError as e:
        out = _Out("not_estimable", None, str(e))
    values = _plain(out.values)

    return {"id": rid, "item": ITEM_OF[rid], "status": out.status, "values": values, "reason": out.reason,

def _restrict(design, mask):

    mask = np.asarray(mask, dtype=bool)
    if not mask.any():

def _derive(design, new_cols, new_names, drop=()):
    """Section 6's derived design: the columns named in `drop` removed, the new ones appended, then the rank check.
    Appending is np.column_stack([design.X, *new]) as section 6 writes it; a removal uses np.delete, which also keeps
    the C layout (the layout changes the last bits of the fit)."""
    gone = [j for j, n in enumerate(design.names) if n in drop]
    X = np.column_stack([np.delete(design.X, gone, axis=1) if gone else design.X, *new_cols])
    names = tuple(n for n in design.names if n not in drop) + tuple(new_names)
    derived = dataclasses.replace(design, X=X, names=names)
    if np.linalg.matrix_rank(derived.X) < derived.X.shape[1]:
        raise _NotEstimable("rank-deficient design", len(derived.y), int(derived.T))
    return derived

        part = {"status": "estimated", "reason": None, "from": lo, "to": hi, "b": None, "delta_cc": None,
                "f_defined": False, "f": None, "se_delta": None, "ci_delta": None, "fieller": None,
                "t_denominator": None, "n_rows": None, "n_dates": None, "lag": None}
        try:
            J = inference.fit_joint(F, self.spec)
        except inference.DesignError as e:
            part.update(status="not_estimable", reason=str(e))
            return part
        b, d, _ = J.pair(self.h1)
        part.update(n_rows=int(J.n), n_dates=int(J.T), lag=int(J.lag))
        if not (math.isfinite(b) and math.isfinite(d)):
            part.update(status="not_estimable", reason="non-finite estimate")
            return part
        part.update(b=float(b), delta_cc=float(d), f_defined=bool(d != 0.0))
        if part["f_defined"]:
            fr = encode_fraction(inference.fraction_unpriced(J, self.h1))
            if not (math.isfinite(fr["f"]) and math.isfinite(fr["se_delta"])):
                part.update(status="not_estimable", reason="non-finite estimate", f_defined=False)
                return part
            part.update({k: fr[k] for k in ("f", "se_delta", "ci_delta", "fieller", "t_denominator")})
        return part

        cols, names = columns_of(without)
        return self.with_without(name, without, _derive(without, cols, names, drop), extra)

def infer(dev, events, confirm=None, *, spec=None) -> dict:
    """The output document of section 4 for the dev frame alone (dev mode) or with the confirmation frame."""
    p_dev = blind.provenance_of(dev)
    if confirm is None:
        if p_dev not in (blind.SYNTHETIC, blind.DEV_REAL):
            raise ValueError(f"dev mode needs a synthetic or dev_real frame, got provenance {p_dev!r}")
    else:
        p_conf = blind.provenance_of(confirm)
        if (p_dev, p_conf) not in ((blind.SYNTHETIC, blind.SYNTHETIC), (blind.DEV_REAL, blind.CONFIRM_REAL)):
            raise ValueError(f"refused provenance pair ({p_dev!r}, {p_conf!r})")
    blind.require_unblinded(dev)
    if confirm is not None:
        blind.require_unblinded(confirm)
    spec = spec or Spec.load()
    cfg = load_config()

    if confirm is not None:
        mode = "confirmation"
        J = pd.concat([dev, confirm], ignore_index=True)
        J.attrs = {"provenance": blind.provenance_of(confirm)}
    else:
        mode = "dev"
        J = dev.copy()
    T = targets.add_masks(targets.add_targets(J, spec), events, spec)
    T250 = targets.add_targets(J, spec, method="rolling", window=250)
    T500 = targets.add_targets(J, spec, method="rolling", window=500)
    periods = {}
    if mode == "confirmation":
        keep = (T["date"] >= confirm["date"].min()).to_numpy()
        C, C250, C500 = _rows_of(T, keep), _rows_of(T250, keep), _rows_of(T500, keep)
        for part in ("confirmatory", "descriptive"):
            lo, hi = (pd.Timestamp(cfg["f"][part][k]) for k in ("from", "to"))
            periods[part] = _rows_of(C, ((C["date"] >= lo) & (C["date"] <= hi)).to_numpy())
    else:
        C, C250, C500 = T, T250, T500

    run = _Run(spec, cfg, mode, C, C250, C500, periods)
    rows: dict = {}
    rows["O1"] = _row("O1", run.o1)
    rows["R1"] = _row("R1", run.r1)
    rows["R2"] = _row("R2", run.r2)
    rows["R3"] = _row("R3", run.r3)
    rows["R4"] = _row("R4", lambda: run.r4(rows["O1"]))
    rows["R5"] = _row("R5", run.r5)
    rows["O2"] = _row("O2", run.o2)
    for rid, term in zip(SECONDARY_ROWS, inference.SECONDARY_TERMS):
        rows[rid] = _row(rid, lambda term=term: run.secondary(term))
    rows["R6"] = _row("R6", lambda: run.r6([rows[i] for i in SECONDARY_ROWS]))
    for rid in ("O7", "O8", "O9", "O10", "O11", "O12", "O13", "O14") + WITH_WITHOUT_ROWS:
        rows[rid] = _row(rid, getattr(run, rid.lower()))
    for rid, name, extra in EXCLUSION_ROWS:
        rows[rid] = _row(rid, lambda name=name, extra=extra: run.exclusion(name, extra))
    rows["O27"] = _row("O27", run.o27)
    rows["O28"] = _row("O28", run.o28)
    rows["R7"] = _row("R7", lambda: run.r7(rows["O28"]))
    rows["R8"] = _row("R8", run.r8)

    return {"module": "confirm_inference", "spec_version": 1, "config_sha256": CONFIG_SHA256, "mode": mode,
            "provenance": J.attrs["provenance"],
            "dates": {"first": _iso(C["date"].min()), "last": _iso(C["date"].max()), "n": int(C["date"].nunique())},
            "rows": {rid: rows[rid] if ITEM_OF[rid] == "P4-4a" else _pending(rid) for rid in ROW_IDS}}

    def test_module_source(self):

        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for a in node.names:
                    self.assertIn(a.name, absolute)
            elif isinstance(node, ast.ImportFrom):
                names = {a.name for a in node.names}
                if node.level == 0:
                    self.assertIn(node.module, absolute)
                elif node.module is None:
                    self.assertLessEqual(names, {"blind", "calib", "inference", "targets"})
                else:
                    self.assertEqual(node.module, "spec")
                    self.assertLessEqual(names, {"Spec", "REPO_ROOT"})
        called = []
        for node in ast.walk(tree):
            if isinstance(node, ast.Call):
                f = node.func
                called.append(f.id if isinstance(f, ast.Name) else f.attr if isinstance(f, ast.Attribute) else None)
        for bad in ("open", "read_csv", "to_csv", "write_text", "write_bytes", "to_parquet", "to_pickle", "to_excel"):
            self.assertNotIn(bad, called)
        self.assertEqual(called.count("read_bytes"), 1)
        for s in ("load_dev", "load_confirm", "load_events", "OFILT_DATA_DIR"):
            self.assertNotIn(s, src)
        functions = {n.name for n in tree.body if isinstance(n, ast.FunctionDef)}
        assigned = {t.id for n in tree.body if isinstance(n, ast.Assign) for t in n.targets if isinstance(t, ast.Name)}

        self.assertLessEqual({"CONFIG_SHA256", "CONFIG_PATH", "ROW_IDS", "ITEM_OF", "STATUSES", "TICKERS"}, assigned)
