def expected_values(msg=MSG):
    lines = open(msg, encoding="utf-8").read().split("\n")
    vals, counts = {}, {}
    for ln in lines[lines.index("## Expected values") + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            counts[m.group(1)] = counts.get(m.group(1), 0) + 1
            vals[m.group(1)] = m.group(2)
    return vals, [k for k, c in counts.items() if c != 1]


class Checker:
    def __init__(self, verdict):
        self.verdict, self.fails = verdict, []

    def __call__(self, ok, label):
        print(("ok   " if ok else "FAIL ") + label)
        if not ok:
            self.fails.append(label)
        return ok

    def finish(self):
        print(f"failed checks: {len(self.fails)}")
        print(f"{self.verdict} CHECK: {not self.fails}")
        return 1 if self.fails else 0


def git(*args):
    return subprocess.run(["git", "-C", REPO, *args], check=True, capture_output=True).stdout


def gtext(*args):
    return git(*args).decode("utf-8").rstrip("\n")


def sha_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha_file(path):
    with open(path, "rb") as fh:
        return hashlib.sha256(fh.read()).hexdigest()


def head_sha(path):
    return sha_bytes(git("show", f"HEAD:{path}"))


def status_lines():
    out = gtext("status", "--porcelain", "--untracked-files=all")
    return out.split("\n") if out else []


def tripwires(check, E, at_head=False):
    """The tags, BLIND_FIT_LOG, RUN_REGISTRY.jsonl (SHA-256 and lines), the spec, the configuration, no ACCESS_LOG."""
    tags = gtext("tag", "-l").split("\n")
    check(sorted(tags) == sorted(E["TAG_LIST"].split(" ")), f"tags == TAG_LIST ({tags})")
    check(gtext("rev-parse", "refs/tags/spec-dev-v1") == E["TAG_OBJECT"], "spec-dev-v1 object == TAG_OBJECT")
    check(gtext("rev-parse", "refs/tags/spec-dev-v1^{commit}") == E["TAG_COMMIT"], "spec-dev-v1 commit == TAG_COMMIT")
    for name, key in (("BLIND_FIT_LOG", "BLIND_FIT_LOG"), ("RUN_REGISTRY.jsonl", "RUN_REGISTRY")):
        b = git("show", f"HEAD:{name}") if at_head else open(os.path.join(REPO, name), "rb").read()
        n = len(b.decode("utf-8").splitlines())
        print(f"{name}{' at HEAD' if at_head else ''}: sha256 {sha_bytes(b)}; lines {n}")
        check(sha_bytes(b) == E[f"{key}_SHA256"] and n == int(E[f"{key}_LINES"]), f"{name} sha256 and lines as expected")
    for path, key in (("spec/analysis_spec_dev_v1.yaml", "SPEC_SHA256"), ("docs/P4_CONFIRM_CONFIG.json", "CONFIG_SHA256")):
        h = head_sha(path) if at_head else sha_file(os.path.join(REPO, path))
        check(h == E[key], f"{path}{' at HEAD' if at_head else ''} == {key}")
    found = []
    for root, dirs, names in os.walk(REPO):
        dirs[:] = [d for d in dirs if not (root == REPO and d == ".git")]
        found += [os.path.relpath(os.path.join(root, n), REPO) for n in dirs + names if n.startswith("ACCESS_LOG")]
    check(found == [], f"no name starting ACCESS_LOG in REPO outside .git ({found})")
