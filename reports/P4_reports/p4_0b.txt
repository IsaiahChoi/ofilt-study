def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def git(*args):
    return subprocess.run(["git", "-C", str(REPO), *args],
                          capture_output=True, text=True, check=True).stdout


def one(pattern, text=None, flags=0):
    found = re.findall(pattern, MSG if text is None else text, flags)
    assert len(found) == 1, (pattern, found)
    return found[0]


def ticks(s):
    """The backticked items of s, in order."""
    return re.findall(r"`([^`]+)`", s)


def show(label, expected, actual):
    ok = expected == actual
    print(f"{label}: expected={expected!r} actual={actual!r} ok={ok}")
    return ok


def prompt():
    """(path, expected SHA-256) of the standing prompt, as this message names it."""
    rel, sha = one(r"standing Pass 4 prompt \(`([^`]+)`, SHA-256 `([0-9a-f]{64})`\)")

def five_new():

    names = ticks(s)
    assert len(names) == 5, names
    return names


def status_block():
    """The PASS4_STATUS.md template: the lines strictly between the two marker lines."""

# P4-0b b: the copy table of step b.1, read from /tmp/p4_0b_msg.txt.
import sys

sys.path.insert(0, "/tmp")
from p4_0b_common import HOME, MSG, REPO  # noqa: E402

HEADER = "| Source | Destination | SHA-256 |"


def rows():
    """[(source Path, destination Path, expected SHA-256 or None)] in table order."""
    from pathlib import Path
    lines = MSG.split("\n")
    assert lines.count(HEADER) == 1
    i = lines.index(HEADER)
    assert set(lines[i + 1].replace("|", "")) == {"-"}, lines[i + 1]
    out = []
    for ln in lines[i + 2:]:
        if not ln.startswith("|"):
            break
        cells = [c.strip() for c in ln.strip().strip("|").split("|")]
        assert len(cells) == 3, ln
        assert all(c.startswith("`") and c.endswith("`") for c in cells[:2]), ln
        src = Path(cells[0][1:-1].replace("$HOME", HOME))
        dst = REPO / cells[1][1:-1]
        if cells[2] == "(printed)":
            sha = None
        else:
            assert cells[2].startswith("`") and cells[2].endswith("`") and len(cells[2]) == 66, ln
            sha = cells[2][1:-1]
        out.append((src, dst, sha))
    assert len(out) == 4, out
    return out

# P4-0b b.2: first two lines, last two non-empty lines, the third line and whole-line counts of each copy.
# Lines are compared after removing only their line endings (\n, and a \r before it).
import re
import sys

sys.path.insert(0, "/tmp")
from p4_0b_common import MSG, REPO, show, ticks  # noqa: E402
from p4_0b_table import rows  # noqa: E402

sec = MSG.split("\n2. By script, of each copy,")[1].split("\n## c. ")[0]
bullets = [ln for ln in sec.split("\n") if ln.startswith("- `reports/")]
print("bullets read from the message:", len(bullets))

oks = [show("bullet paths = the copy table's destinations", dests, [ticks(b)[0] for b in bullets])]


def lines_of(p):
    parts = p.read_bytes().decode("utf-8").split("\n")
    if parts and parts[-1] == "":
        parts.pop()
    return [x[:-1] if x.endswith("\r") else x for x in parts]


for b in bullets:
    m = re.match(r"- `([^`]+)`: first two lines `([^`]+)` and `([^`]+)`; last two non-empty lines `([^`]+)` and `([^`]+)`; "
                 r"third line `([^`]+)`; that whole line occurs exactly (\d+) times?(.*)$", b)
    assert m, b
    rel, f1, f2, l1, l2, third, n3, rest = m.groups()
    extra = re.findall(r"the whole line `([^`]+)` occurs exactly (\d+) times?", rest)
    lines = lines_of(REPO / rel)
    nonempty = [x for x in lines if x != ""]
    print(f"--- {rel}: {len(lines)} lines; last 3 lines as read: {lines[-3:]!r}")
    oks.append(show("  first two lines", [f1, f2], lines[:2]))
    oks.append(show("  last two non-empty lines", [l1, l2], nonempty[-2:]))
    oks.append(show("  third line", third, lines[2]))
    oks.append(show("  occurrences of the third line", int(n3), lines.count(third)))
    for text, n in extra:
        oks.append(show(f"  occurrences of {text!r}", int(n), lines.count(text)))
print("checks:", len(oks), "passed:", sum(oks))
print("B2 CHECK:", all(oks))
