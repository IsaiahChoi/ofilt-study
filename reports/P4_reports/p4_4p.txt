def shape_of(v):

        return f"ndarray{v.shape} {v.dtype}"

    return f"{type(v).__name__} = {v!r}" if isinstance(v, (int, float, str, bool, type(None), np.floating,

        for k, v in x.items():
            lines.append(f"  [{k!r}]: {shape_of(v)}")
        est = {k: x[k] for k in ("estimate", "se", "p") if k in x}
        if est:
            lines.append("  H1 (this dict): " + ", ".join(f"{k} {v:.6g}" for k, v in est.items()))
    elif isinstance(x, pd.DataFrame):
        lines.append(f"  shape {x.shape}; columns: " + ", ".join(f"{c} ({x[c].dtype})" for c in x.columns))
    elif isinstance(x, (np.ndarray, tuple, list)):
        lines.append(f"  {shape_of(x)}")
    else:
        lines.append(f"  {x!r}")
    return "\n".join(lines)


def call(run, label, fn, full=False):
    OUTCOMES.append((run, label))
    t0 = time.perf_counter()
    try:
        res = fn()
        dt = time.perf_counter() - t0
        print(f"CALL {label} | wall {dt:.3f} s\n{describe(res)}")
        if full and isinstance(res, pd.DataFrame):
            with pd.option_context("display.max_rows", 100, "display.max_columns", 20, "display.width", 200):
                print(res.to_string())
        return res
    except Exception as e:   # an outcome, reported
        dt = time.perf_counter() - t0
        print(f"CALL {label} | wall {dt:.3f} s\nEXCEPTION {type(e).__name__}: {e}")
        return None


def restrict(T, lo=None, hi=None):
    m = pd.Series(True, index=T.index)
    if lo is not None:
        m &= T["date"] >= lo
    if hi is not None:
        m &= T["date"] <= hi
    R = T.loc[m]
    R.attrs = dict(T.attrs)
    return R
