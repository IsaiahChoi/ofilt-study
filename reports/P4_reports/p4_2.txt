            assert m.group(1) not in out, m.group(1)

    return out

def ev(name):
    """The value of a label; each label begins exactly one line of the list."""
    sec = MSG.split("\n## Expected values\n")[1].split("\n## ")[0]
    assert sum(ln.startswith(f"- {name} = ") for ln in sec.split("\n")) == 1, name
    return EV[name]


def sha256_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha256(p):
    return sha256_bytes(Path(p).read_bytes())


def git(*args, check=True):

def table(header):
    """The body rows of the message's table whose header line is `header`, as lists of stripped cells."""
    lines = MSG.split("\n")
    assert lines.count(header) == 1, header
    i = lines.index(header)
    assert set(lines[i + 1].replace("|", "")) == {"-"}
    rows = []
    for ln in lines[i + 2:]:
        if not ln.startswith("|"):
            break
        rows.append([c.strip() for c in ln.strip()[1:-1].split("|")])
    return rows

    m = re.fullmatch(r"`([^`]+)`", cell)

def _expected():
    sec = MSG.split("\n## Expected values\n")[1].split("\n## ")[0]
    out = {}
    for ln in sec.split("\n"):
        if ln.startswith("- "):
            m = re.fullmatch(r"- ([A-Z0-9_]+) = (.+)", ln)
            assert m, ln
            assert m.group(1) not in out, m.group(1)
            out[m.group(1)] = m.group(2)
    return out


EV = _expected()


def ev(name):
    """The value of a label; each label begins exactly one line of the list."""
    sec = MSG.split("\n## Expected values\n")[1].split("\n## ")[0]
    assert sum(ln.startswith(f"- {name} = ") for ln in sec.split("\n")) == 1, name
    return EV[name]


def sha256_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha256(p):
    return sha256_bytes(Path(p).read_bytes())


def git(*args, check=True):
    return subprocess.run(["git", "-C", str(REPO), *args], capture_output=True, text=True, check=check)


def table(header):
    """The body rows of the message's table whose header line is `header`, as lists of stripped cells."""
    lines = MSG.split("\n")
    assert lines.count(header) == 1, header
    i = lines.index(header)
    assert set(lines[i + 1].replace("|", "")) == {"-"}
    rows = []
    for ln in lines[i + 2:]:
        if not ln.startswith("|"):
            break
        rows.append([c.strip() for c in ln.strip()[1:-1].split("|")])
    return rows


def tick(cell):
    m = re.fullmatch(r"`([^`]+)`", cell)
    return m.group(1) if m else None


def kit_rows():
    """[(kit file name, label, destination or None)] of step b's table."""
    out = []
    for name, label, dest in table("| Kit file | SHA-256 | Destination in REPO |"):
        out.append((tick(name), label, tick(dest)))
    return out


def record_rows():
    """[(name, destination, label, line checks)] of step c's table."""
    return [(tick(n), tick(d), lab, lc) for n, d, lab, lc in table("| Name | Destination | SHA-256 | Line checks |")]


def destinations():
    return [d for _, _, d in kit_rows() if d] + [d for _, d, _, _ in record_rows()]


def show(label, expected, actual):

# P4-2 d.2 and e.1 helper: read the kit's edits file by the rules in its header, and apply a list of its edits to a
# text. Reads the hashed kit file only (its SHA-256 is checked first); writes nothing.
import re
import sys
from pathlib import Path

sys.path.insert(0, "/tmp")
from p4_2_common import KIT, ev, sha256  # noqa: E402

EDITS_FILE = KIT / "p4_2_STATUS_EDITS.txt"
MARK = re.compile(r"^@@@ (EDIT \d\d|FINAL) (OLD|NEW|END)$")


def read_edits():
    """(placeholder, {name: (old, new)}) in file order; the file's SHA-256 must be KIT_EDITS_SHA256."""
    assert sha256(EDITS_FILE) == ev("KIT_EDITS_SHA256"), "edits file SHA-256"
    lines = EDITS_FILE.read_bytes().decode("utf-8").split("\n")
    if lines and lines[-1] == "":
        lines.pop()
    header = []
    k = 0
    while k < len(lines) and not MARK.match(lines[k]):
        header.append(lines[k])
        k += 1
    ph = re.findall(r"each occurrence of the placeholder (@@[A-Z]+@@) is replaced", " ".join(header))
    assert len(ph) == 1, ph
    edits = {}
    while k < len(lines):
        m = MARK.match(lines[k])
        assert m and m.group(2) == "OLD", (k, lines[k])
        name = m.group(1)
        assert name not in edits, name
        j = k + 1
        while not MARK.match(lines[j]):
            j += 1
        assert MARK.match(lines[j]).groups() == (name, "NEW"), (j, lines[j])
        e = j + 1
        while not MARK.match(lines[e]):
            e += 1
        assert MARK.match(lines[e]).groups() == (name, "END"), (e, lines[e])
        edits[name] = ("\n".join(lines[k + 1:j]), "\n".join(lines[j + 1:e]))
        k = e + 1
    return ph[0], edits


def apply(text, names, date):
    """Apply the named edits in order; each OLD must occur exactly once when applied. Returns (text, log lines)."""
    ph, edits = read_edits()
    log = []
    for name in names:
        old, new = edits[name]
        new = new.replace(ph, date)
        n = text.count(old)
        log.append(f"{name}: OLD occurs {n} time(s); OLD {old.count(chr(10)) + 1} line(s) -> NEW {new.count(chr(10)) + 1} line(s)")
        if n != 1:
            return None, log
        text = text.replace(old, new, 1)
    return text, log
