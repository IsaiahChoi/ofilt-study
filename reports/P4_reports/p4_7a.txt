def expected_values(msg=MSG):
    lines = open(msg, encoding="utf-8").read().split("\n")
    vals, counts = {}, {}
    for ln in lines[lines.index("## Expected values") + 1:]:
        if ln.startswith("## "):
            break
        m = re.match(r"^- ([A-Z0-9_]+) = (.*)$", ln)
        if m:
            counts[m.group(1)] = counts.get(m.group(1), 0) + 1
            vals[m.group(1)] = m.group(2)
    return vals, [k for k, c in counts.items() if c != 1]


class Checker:
    def __init__(self, verdict):
        self.verdict, self.fails = verdict, []

    def __call__(self, ok, label):
        print(("ok   " if ok else "FAIL ") + label)
        if not ok:
            self.fails.append(label)
        return ok

    def finish(self):
        print(f"failed checks: {len(self.fails)}")
        print(f"{self.verdict} CHECK: {not self.fails}")
        return 1 if self.fails else 0


def git(*args):
    return subprocess.run(["git", "-C", REPO, *args], check=True, capture_output=True).stdout


def gtext(*args):
    return git(*args).decode("utf-8").rstrip("\n")


def sha_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha_file(path):
    with open(path, "rb") as fh:
        return hashlib.sha256(fh.read()).hexdigest()


def head_sha(path):
    return sha_bytes(git("show", f"HEAD:{path}"))


def status_lines():
    out = gtext("status", "--porcelain", "--untracked-files=all")
    return out.split("\n") if out else []
